32/// FastAppendAction is a transaction action for fast append data files to the table.
33#[derive(Clone)]
34pub struct FastAppendAction {
35    check_duplicate: bool,
36    // below are properties used to create SnapshotProducer when commit
37    commit_uuid: Option<Uuid>,
38    snapshot_properties: HashMap<String, String>,
39    added_data_files: Vec<DataFile>,
40}
41
42impl FastAppendAction {
43    pub(crate) fn new() -> Self {
44        Self {
45            check_duplicate: true,
46            commit_uuid: None,
47            snapshot_properties: HashMap::default(),
48            added_data_files: vec![],
49        }
50    }
51
52    /// Set whether to check duplicate files
53    pub fn with_check_duplicate(mut self, v: bool) -> Self {
54        self.check_duplicate = v;
55        self
56    }
57
58    /// Add data files to the snapshot.
59    pub fn add_data_files(mut self, data_files: impl IntoIterator<Item = DataFile>) -> Self {
60        self.added_data_files.extend(data_files);
61        self
62    }
63
64    /// Set commit UUID for the snapshot.
65    pub fn set_commit_uuid(mut self, commit_uuid: Uuid) -> Self {
66        self.commit_uuid = Some(commit_uuid);
67        self
68    }
69
70    /// Set snapshot summary properties.
71    pub fn set_snapshot_properties(mut self, snapshot_properties: HashMap<String, String>) -> Self {
72        self.snapshot_properties = snapshot_properties;
73        self
74    }
75
76    /// Collapse files sharing a path to their first occurrence, so a single
77    /// manifest never references the same file twice. Always runs (unlike the
78    /// `check_duplicate`-gated cross-snapshot check) since it is in-memory only.
79    fn dedupe_added_files(&self) -> Vec<DataFile> {
80        let mut seen = HashSet::with_capacity(self.added_data_files.len());
81        self.added_data_files
82            .iter()
83            .filter(|data_file| seen.insert(data_file.file_path.as_str()))
84            .cloned()
85            .collect()
86    }
87}
88
89#[async_trait]
90impl TransactionAction for FastAppendAction {
91    // TODO: replace with a persistent `SimpleSnapshotProducer` once snapshot
92    // production is migrated (stateful transaction RFC, milestone 2).
93    type State = ();
94
95    fn new_state(&self) -> Self::State {}
96
97    async fn commit(&self, _state: &mut (), table: &Table) -> Result<ActionCommit> {
98        let snapshot_producer = SnapshotProducer::new(
99            table,
100            self.commit_uuid.unwrap_or_else(Uuid::now_v7),
101            self.snapshot_properties.clone(),
102            self.dedupe_added_files(),
103            vec![],
104        );
105
106        // validate added files
107        snapshot_producer.validate_added_data_files()?;
108
109        // Checks duplicate files
110        if self.check_duplicate {
111            snapshot_producer.validate_duplicate_files().await?;
112        }
113
114        snapshot_producer
115            .commit(FastAppendOperation, DefaultManifestProcess)
116            .await
117    }
118
119    async fn cleanup(self, _state: (), _table: &Table, _status: CommitStatus) {
120        // Stateless for now: generated manifests are not yet tracked for
121        // terminal cleanup (stateful transaction RFC, milestone 2/3).
122    }
123}
124
125struct FastAppendOperation;
126
127impl SnapshotProduceOperation for FastAppendOperation {
128    fn operation(&self) -> Operation {
129        Operation::Append
130    }
131
132    async fn delete_entries(
133        &self,
134        _snapshot_produce: &SnapshotProducer<'_>,
135    ) -> Result<Vec<ManifestEntry>> {
136        Ok(vec![])
137    }
138
139    async fn existing_manifest(
140        &self,
141        snapshot_produce: &SnapshotProducer<'_>,
142    ) -> Result<Vec<ManifestFile>> {
143        let Some(snapshot) = snapshot_produce.table.metadata().current_snapshot() else {
144            return Ok(vec![]);
145        };
146
147        let manifest_list = snapshot_produce
148            .table
149            .manifest_list_reader(snapshot)
150            .load()
151            .await?;
152
153        Ok(manifest_list
154            .entries()
155            .iter()
156            .filter(|entry| {
157                // Keep delete-only manifests too: they record which files were removed and
158                // must persist across snapshots until `expire_snapshots` cleans them up.
159                // Dropping them lets the removed files reappear as live data (see #2148).
160                entry.has_added_files() || entry.has_existing_files() || entry.has_deleted_files()
161            })
162            .cloned()
163            .collect())
164    }
165}

405    async fn test_fast_append_writes_encrypted_manifest() {
406        let table = make_encrypted_table().await;
407        assert!(
408            table.encryption_manager().is_some(),
409            "fixture table should have an EncryptionManager"
410        );
411
412        let new_file = DataFileBuilder::default()
413            .content(DataContentType::Data)
414            .file_path("memory:///table/data/00000.parquet".to_string())
415            .file_format(DataFileFormat::Parquet)
416            .partition(Struct::empty())
417            .record_count(100)
418            .file_size_in_bytes(4096)
419            .partition_spec_id(table.metadata().default_partition_spec_id())
420            .build()
421            .unwrap();
422
423        let tx = Transaction::new(&table);
424        let action = tx.fast_append().add_data_files(vec![new_file]);
425        let mut action_commit = commit_with_fresh_state(action, &table).await.unwrap();
426        let updates = action_commit.take_updates();
427
428        let new_snapshot: SnapshotRef = updates
429            .iter()
430            .find_map(|u| match u {
431                TableUpdate::AddSnapshot { snapshot } => Some(SnapshotRef::new(snapshot.clone())),
432                _ => None,
433            })
434            .expect("a fast append should emit an AddSnapshot update");
435
436        let manifest_list_key_metadata = table
437            .encryption_manager()
438            .unwrap()
439            .decrypt_manifest_list_key_metadata(new_snapshot.encryption_key_id().unwrap())
440            .await
441            .unwrap();
442        let manifest_list_size = table
443            .file_io()
444            .new_input(new_snapshot.manifest_list())
445            .unwrap()
446            .metadata()
447            .await
448            .unwrap()
449            .size;
450        assert_eq!(
451            manifest_list_key_metadata.file_length(),
452            Some(manifest_list_size)
453        );
454
455        let manifest_list = table
456            .manifest_list_reader(&new_snapshot)
457            .load()
458            .await
459            .unwrap();
460        let manifest_file = manifest_list
461            .entries()
462            .iter()
463            .find(|m| m.added_files_count.unwrap_or(0) > 0)
464            .expect("new snapshot should carry the appended data manifest");
465
466        // The manifest list entry must carry decodable key metadata.
467        let key_metadata_bytes = manifest_file
468            .key_metadata
469            .as_ref()
470            .expect("encrypted manifest must record key metadata");
471        let key_metadata = StandardKeyMetadata::decode(key_metadata_bytes)
472            .expect("recorded key metadata must decode as StandardKeyMetadata");
473        let manifest_size = table
474            .file_io()
475            .new_input(&manifest_file.manifest_path)
476            .unwrap()
477            .metadata()
478            .await
479            .unwrap()
480            .size;
481        assert_eq!(key_metadata.file_length(), Some(manifest_size));
482        assert_eq!(manifest_file.manifest_length, manifest_size as i64);
483
484        // The reader self-decrypts using the recorded key metadata and must
485        // recover the entry we appended. Because the read goes through the
486        // decryption path, this succeeding also proves the bytes
487        // on disk were genuinely encrypted (not silently written as plaintext).
488        let manifest = table.manifest_reader().read(manifest_file).await.unwrap();
489        assert_eq!(manifest.entries().len(), 1);
490        assert_eq!(
491            manifest.entries()[0].data_file().file_path(),
492            "memory:///table/data/00000.parquet"
493        );
494    }

726    #[tokio::test]
727    async fn test_snapshot_properties_cannot_override_computed_metrics() {
728        // A user-supplied snapshot property must not shadow a computed metric key
729        // such as `added-data-files`. Matching iceberg-java, the computed value
730        // wins, so the summary reflects the real count and a bad value can neither
731        // corrupt the summary nor panic total computation (see #2184-adjacent fix).
732        let table = make_v2_minimal_table();
733        let tx = Transaction::new(&table);
734
735        let mut snapshot_properties = HashMap::new();
736        // Both a benign-but-wrong value and a non-integer value collide with
737        // computed metric keys; neither should reach the final summary.
738        snapshot_properties.insert("added-data-files".to_string(), "9999".to_string());
739        snapshot_properties.insert("added-records".to_string(), "not-a-number".to_string());
740
741        let data_file = DataFileBuilder::default()
742            .content(DataContentType::Data)
743            .file_path("test/1.parquet".to_string())
744            .file_format(DataFileFormat::Parquet)
745            .file_size_in_bytes(100)
746            .record_count(1)
747            .partition_spec_id(table.metadata().default_partition_spec_id())
748            .partition(Struct::from_iter([Some(Literal::long(300))]))
749            .build()
750            .unwrap();
751
752        let action = tx
753            .fast_append()
754            .set_snapshot_properties(snapshot_properties)
755            .add_data_files(vec![data_file]);
756        // Must not panic during total computation.
757        let mut action_commit = commit_with_fresh_state(action, &table).await.unwrap();
758        let updates = action_commit.take_updates();
759
760        let new_snapshot = if let TableUpdate::AddSnapshot { snapshot } = &updates[0] {
761            snapshot
762        } else {
763            unreachable!()
764        };
765        let props = &new_snapshot.summary().additional_properties;
766
767        // Computed metric wins over the user's colliding values.
768        assert_eq!(
769            props.get("added-data-files").unwrap(),
770            "1",
771            "computed added-data-files must override the user-supplied value"
772        );
773        assert_eq!(
774            props.get("added-records").unwrap(),
775            "1",
776            "computed added-records must override the user-supplied non-integer value"
777        );
778    }

892    async fn test_fast_append() {
893        let table = make_v2_minimal_table();
894        let tx = Transaction::new(&table);
895        let action = tx.fast_append();
896
897        let data_file = DataFileBuilder::default()
898            .content(DataContentType::Data)
899            .file_path("test/3.parquet".to_string())
900            .file_format(DataFileFormat::Parquet)
901            .file_size_in_bytes(100)
902            .record_count(1)
903            .partition_spec_id(table.metadata().default_partition_spec_id())
904            .partition(Struct::from_iter([Some(Literal::long(300))]))
905            .build()
906            .unwrap();
907
908        let action = action.add_data_files(vec![data_file.clone()]);
909        let mut action_commit = commit_with_fresh_state(action, &table).await.unwrap();
910        let updates = action_commit.take_updates();
911        let requirements = action_commit.take_requirements();
912
913        // check updates and requirements
914        assert!(
915            matches!((&updates[0],&updates[1]), (TableUpdate::AddSnapshot { snapshot },TableUpdate::SetSnapshotRef { reference,ref_name }) if snapshot.snapshot_id() == reference.snapshot_id && ref_name == MAIN_BRANCH)
916        );
917        assert_eq!(
918            vec![
919                TableRequirement::UuidMatch {
920                    uuid: table.metadata().uuid()
921                },
922                TableRequirement::RefSnapshotIdMatch {
923                    r#ref: MAIN_BRANCH.to_string(),
924                    snapshot_id: table.metadata().current_snapshot_id
925                }
926            ],
927            requirements
928        );
929
930        // check manifest list
931        let new_snapshot: SnapshotRef = if let TableUpdate::AddSnapshot { snapshot } = &updates[0] {
932            SnapshotRef::new(snapshot.clone())
933        } else {
934            unreachable!()
935        };
936        let manifest_list = table
937            .manifest_list_reader(&new_snapshot)
938            .load()
939            .await
940            .unwrap();
941        assert_eq!(1, manifest_list.entries().len());
942        assert_eq!(
943            manifest_list.entries()[0].sequence_number,
944            new_snapshot.sequence_number()
945        );
946
947        // check manifest
948        let manifest = table
949            .manifest_reader()
950            .read(&manifest_list.entries()[0])
951            .await
952            .unwrap();
953        assert_eq!(1, manifest.entries().len());
954        assert_eq!(
955            new_snapshot.sequence_number(),
956            manifest.entries()[0]
957                .sequence_number()
958                .expect("Inherit sequence number by load manifest")
959        );
960
961        assert_eq!(
962            new_snapshot.snapshot_id(),
963            manifest.entries()[0].snapshot_id().unwrap()
964        );
965        assert_eq!(data_file, *manifest.entries()[0].data_file());
