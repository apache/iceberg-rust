32/// FastAppendAction is a transaction action for fast append data files to the table.
33#[derive(Clone)]
34pub struct FastAppendAction {
35    check_duplicate: bool,
36    /// Overrides the commit UUID of the producer created in `new_state`.
37    commit_uuid: Option<Uuid>,
38    snapshot_properties: HashMap<String, String>,
39    /// Free of duplicate paths; see [`Self::add_data_files`].
40    added_data_files: Vec<DataFile>,
41    /// Paths already in `added_data_files`.
42    added_paths: HashSet<String>,
43}
44
45impl FastAppendAction {
46    pub(crate) fn new() -> Self {
47        Self {
48            check_duplicate: true,
49            commit_uuid: None,
50            snapshot_properties: HashMap::default(),
51            added_data_files: vec![],
52            added_paths: HashSet::new(),
53        }
54    }
55
56    /// Set whether to check duplicate files
57    pub fn with_check_duplicate(mut self, v: bool) -> Self {
58        self.check_duplicate = v;
59        self
60    }
61
62    /// Add data files to the snapshot.
63    ///
64    /// A path added more than once keeps its first occurrence, so a manifest
65    /// never references the same file twice. This is independent of the
66    /// `check_duplicate`-gated check against files already in the table.
67    pub fn add_data_files(mut self, data_files: impl IntoIterator<Item = DataFile>) -> Self {
68        for data_file in data_files {
69            if self.added_paths.insert(data_file.file_path.clone()) {
70                self.added_data_files.push(data_file);
71            }
72        }
73        self
74    }
75
76    /// Set commit UUID for the snapshot.
77    pub fn set_commit_uuid(mut self, commit_uuid: Uuid) -> Self {
78        self.commit_uuid = Some(commit_uuid);
79        self
80    }
81
82    /// Set snapshot summary properties.
83    pub fn set_snapshot_properties(mut self, snapshot_properties: HashMap<String, String>) -> Self {
84        self.snapshot_properties = snapshot_properties;
85        self
86    }
87
88    /// Checks the parts of this action that do not depend on the table.
89    fn validate(&self) -> Result<()> {
90        // TODO: Allowing snapshot property setup with no added data files is a workaround.
91        // We should clean it up after all necessary actions are supported.
92        // For details, please refer to https://github.com/apache/iceberg-rust/issues/1548
93        if self.added_data_files.is_empty() && self.snapshot_properties.is_empty() {
94            return Err(Error::new(
95                ErrorKind::PreconditionFailed,
96                "No added data files or added snapshot properties found when write a manifest file",
97            ));
98        }
99        if let Some(data_file) = self
100            .added_data_files
101            .iter()
102            .find(|f| f.content_type() != DataContentType::Data)
103        {
104            return Err(invalid_data!(
105                "Only data content type is allowed for fast append, got {:?} for {}",
106                data_file.content_type(),
107                data_file.file_path()
108            ));
109        }
110        Ok(())
111    }
112}

114#[async_trait]
115impl TransactionAction for FastAppendAction {
116    type State = SimpleSnapshotProducer;
117
118    fn new_state(&self) -> SimpleSnapshotProducer {
119        SimpleSnapshotProducer::new(AppendIntent {
120            commit_uuid: self.commit_uuid,
121            snapshot_properties: self.snapshot_properties.clone(),
122            data_files: self.added_data_files.clone(),
123        })
124    }
125
126    async fn commit(
127        &self,
128        state: &mut SimpleSnapshotProducer,
129        table: &Table,
130    ) -> Result<ActionCommit> {
131        self.validate()?;
132        if self.check_duplicate {
133            validate_files_not_live(table, &self.added_data_files).await?;
134        }
135        state.apply(table).await
136    }
137
138    async fn cleanup(self, _state: SimpleSnapshotProducer, _table: &Table, _status: CommitStatus) {
139        // TODO(#3417): remove the producer's uncommitted artifacts once terminal
140        // cleanup is wired up.
141    }
142}

386    async fn test_fast_append_writes_encrypted_manifest() {
387        let table = make_encrypted_table().await;
388        assert!(
389            table.encryption_manager().is_some(),
390            "fixture table should have an EncryptionManager"
391        );
392
393        let new_file = DataFileBuilder::default()
394            .content(DataContentType::Data)
395            .file_path("memory:///table/data/00000.parquet".to_string())
396            .file_format(DataFileFormat::Parquet)
397            .partition(Struct::empty())
398            .record_count(100)
399            .file_size_in_bytes(4096)
400            .partition_spec_id(table.metadata().default_partition_spec_id())
401            .build()
402            .unwrap();
403
404        let tx = Transaction::new(&table);
405        let action = tx.fast_append().add_data_files(vec![new_file]);
406        let mut action_commit = commit_with_fresh_state(action, &table).await.unwrap();
407        let updates = action_commit.take_updates();
408
409        let new_snapshot: SnapshotRef = updates
410            .iter()
411            .find_map(|u| match u {
412                TableUpdate::AddSnapshot { snapshot } => Some(SnapshotRef::new(snapshot.clone())),
413                _ => None,
414            })
415            .expect("a fast append should emit an AddSnapshot update");
416
417        let manifest_list_key_metadata = table
418            .encryption_manager()
419            .unwrap()
420            .decrypt_manifest_list_key_metadata(new_snapshot.encryption_key_id().unwrap())
421            .await
422            .unwrap();
423        let manifest_list_size = table
424            .file_io()
425            .new_input(new_snapshot.manifest_list())
426            .unwrap()
427            .metadata()
428            .await
429            .unwrap()
430            .size;
431        assert_eq!(
432            manifest_list_key_metadata.file_length(),
433            Some(manifest_list_size)
434        );
435
436        let manifest_list = table
437            .manifest_list_reader(&new_snapshot)
438            .load()
439            .await
440            .unwrap();
441        let manifest_file = manifest_list
442            .entries()
443            .iter()
444            .find(|m| m.added_files_count.unwrap_or(0) > 0)
445            .expect("new snapshot should carry the appended data manifest");
446
447        // The manifest list entry must carry decodable key metadata.
448        let key_metadata_bytes = manifest_file
449            .key_metadata
450            .as_ref()
451            .expect("encrypted manifest must record key metadata");
452        let key_metadata = StandardKeyMetadata::decode(key_metadata_bytes)
453            .expect("recorded key metadata must decode as StandardKeyMetadata");
454        let manifest_size = table
455            .file_io()
456            .new_input(&manifest_file.manifest_path)
457            .unwrap()
458            .metadata()
459            .await
460            .unwrap()
461            .size;
462        assert_eq!(key_metadata.file_length(), Some(manifest_size));
463        assert_eq!(manifest_file.manifest_length, manifest_size as i64);
464
465        // The reader self-decrypts using the recorded key metadata and must
466        // recover the entry we appended. Because the read goes through the
467        // decryption path, this succeeding also proves the bytes
468        // on disk were genuinely encrypted (not silently written as plaintext).
469        let manifest = table.manifest_reader().read(manifest_file).await.unwrap();
470        assert_eq!(manifest.entries().len(), 1);
471        assert_eq!(
472            manifest.entries()[0].data_file().file_path(),
473            "memory:///table/data/00000.parquet"
474        );
475    }

754            .build()
755            .unwrap();
756
757        let action = tx
758            .fast_append()
759            .set_snapshot_properties(snapshot_properties)
760            .add_data_files(vec![data_file]);

1029    /// Runs the concurrent-append rebase scenario on `s0` and returns the
1030    /// snapshots committed by writer B (S1) and writer A (S2).
1031    async fn rebase_onto_concurrent_commit(s0: Table) -> (SnapshotRef, SnapshotRef) {
1032        // `TableCommit::apply` derives the next metadata location from the
1033        // current one, so it needs a well-formed name.
1034        let s0 = with_retry_properties(s0, "3").with_metadata_location(
1035            "memory:///table/metadata/00000-9c12d441-03fe-4693-9a96-a0705ddf69c1.metadata.json"
1036                .to_string(),
1037        );
1038        let data_file = |path: &str| {
1039            DataFileBuilder::default()
1040                .content(DataContentType::Data)
1041                .file_path(path.to_string())
1042                .file_format(DataFileFormat::Parquet)
1043                .file_size_in_bytes(100)
1044                .record_count(1)
1045                .partition_spec_id(s0.metadata().default_partition_spec_id())
1046                .partition(Struct::from_iter([Some(Literal::long(300))]))
1047                .build()
1048                .unwrap()
1049        };
1050
1051        // Writer B commits S1 on top of S0.
1052        let mut b_commit = commit_with_fresh_state(
1053            FastAppendAction::new().add_data_files(vec![data_file("test/b.parquet")]),
1054            &s0,
1055        )
1056        .await
1057        .unwrap();
1058        let s1 = TableCommit::builder()
1059            .ident(s0.identifier().clone())
1060            .updates(b_commit.take_updates())
1061            .requirements(b_commit.take_requirements())
1062            .build()
1063            .apply(s0.clone())
1064            .unwrap();
1065        let s1_snapshot = s1.metadata().current_snapshot().unwrap().clone();
1066
1067        // The catalog serves S0 to A's first attempt and S1 afterwards; every
1068        // update is validated against S1, so the first attempt's
1069        // `RefSnapshotIdMatch { snapshot_id: None }` fails with a real conflict.
1070        let mut mock_catalog = MockCatalog::new();
1071        let loads = AtomicU32::new(0);
1072        let (load_s0, load_s1) = (s0.clone(), s1.clone());
1073        mock_catalog
1074            .expect_load_table()
1075            .times(2)
1076            .returning_st(move |_| {
1077                let table = if loads.fetch_add(1, Ordering::SeqCst) == 0 {
1078                    load_s0.clone()
1079                } else {
1080                    load_s1.clone()
1081                };
1082                Box::pin(async move { Ok(table) })
1083            });
1084        let apply_s1 = s1.clone();
1085        mock_catalog
1086            .expect_update_table()
1087            .times(2)
1088            .returning_st(move |commit| {
1089                let s1 = apply_s1.clone();
1090                Box::pin(async move { commit.apply(s1) })
1091            });
1092
1093        // Writer A starts from S0.
1094        let a_uuid = Uuid::now_v7();
1095        let tx = Transaction::new(&s0);
1096        let tx = tx
1097            .fast_append()
1098            .set_commit_uuid(a_uuid)
1099            .add_data_files(vec![data_file("test/a.parquet")])
1100            .apply(tx)
1101            .unwrap();
1102        let s2 = tx.commit(&mock_catalog).await.unwrap();
1103        let s2_snapshot = s2.metadata().current_snapshot().unwrap().clone();
1104
1105        // S2 is rebased onto S1, not S0.
1106        assert_eq!(
1107            s2_snapshot.parent_snapshot_id(),
1108            Some(s1_snapshot.snapshot_id())
1109        );
1110        assert_eq!(
1111            s2_snapshot.sequence_number(),
1112            s1_snapshot.sequence_number() + 1
1113        );
1114        // The committed manifest list is A's second attempt.
1115        assert!(
1116            s2_snapshot
1117                .manifest_list()
1118                .ends_with(&format!("-1-{a_uuid}.avro")),
1119            "unexpected manifest list path {}",
1120            s2_snapshot.manifest_list()
1121        );
1122
1123        // S2 carries B's manifest forward and adds A's. A's manifest is the one
1124        // written during the first attempt (counter 0), reused after rebase.
1125        let manifest_list = s2.manifest_list_reader(&s2_snapshot).load().await.unwrap();
1126        assert_eq!(manifest_list.entries().len(), 2);
1127        let b_manifest = manifest_list
1128            .entries()
1129            .iter()
1130            .find(|m| m.added_snapshot_id == s1_snapshot.snapshot_id())
1131            .expect("S2 should carry B's manifest forward");
1132        let a_manifest = manifest_list
1133            .entries()
1134            .iter()
1135            .find(|m| m.added_snapshot_id == s2_snapshot.snapshot_id())
1136            .expect("S2 should add A's manifest");
1137        assert!(
1138            a_manifest
1139                .manifest_path
1140                .ends_with(&format!("{a_uuid}-m0.avro")),
1141            "unexpected manifest path {}",
1142            a_manifest.manifest_path
1143        );
1144        assert_eq!(a_manifest.sequence_number, s2_snapshot.sequence_number());
1145        assert_eq!(b_manifest.sequence_number, s1_snapshot.sequence_number());
1146
1147        // Summary totals are computed against S1, not S0.
1148        let summary = &s2_snapshot.summary().additional_properties;
1149        assert_eq!(summary.get("added-data-files").unwrap(), "1");
1150        assert_eq!(summary.get("total-data-files").unwrap(), "2");
1151        assert_eq!(summary.get("total-records").unwrap(), "2");
1152
1153        (s1_snapshot, s2_snapshot)
1154    }
