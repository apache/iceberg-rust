28/// Represents a sort field whose construction and validation are deferred until commit time.
29/// This avoids the need to pass a `Table` reference into methods like `asc` or `desc` when
30/// adding sort orders.
31#[derive(Debug, PartialEq, Eq, Clone)]
32struct PendingSortField {
33    name: String,
34    transform: Transform,
35    direction: SortDirection,
36    null_order: NullOrder,
37}
38
39impl PendingSortField {
40    fn to_sort_field(&self, schema: &SchemaRef) -> Result<SortField> {
41        let field_id = schema.field_id_by_name(self.name.as_str()).ok_or_else(|| {
42            Error::new(
43                ErrorKind::DataInvalid,
44                format!("Cannot find field {} in table schema", self.name),
45            )
46        })?;
47
48        // This action intentionally refuses to author all-null (void) sort keys;
49        // this policy does not restrict reading or preserving existing metadata.
50        if matches!(self.transform, Transform::Unknown | Transform::Void) {
51            return Err(Error::new(
52                ErrorKind::DataInvalid,
53                format!("Cannot sort by transform {}", self.transform),
54            ));
55        }
56
57        Ok(SortField::builder()
58            .source_id(field_id)
59            .transform(self.transform)
60            .direction(self.direction)
61            .null_order(self.null_order)
62            .build())
63    }
64}
65
66/// Transaction action for replacing sort order.
67pub struct ReplaceSortOrderAction {
68    pending_sort_fields: Vec<PendingSortField>,
69}
70
71impl ReplaceSortOrderAction {
72    pub(crate) fn new() -> Self {
73        ReplaceSortOrderAction {
74            pending_sort_fields: vec![],
75        }
76    }
77
78    /// Adds a field for sorting in ascending order, sorting by the column's raw value
79    /// (an identity transform). To sort by a transform of the column instead (e.g.
80    /// `bucket[N]`, `year`, `truncate[W]`), use [`Self::asc_with_transform`].
81    pub fn asc(self, name: &str, null_order: NullOrder) -> Self {
82        self.asc_with_transform(name, Transform::Identity, null_order)
83    }
84
85    /// Adds a field for sorting in descending order, sorting by the column's raw value
86    /// (an identity transform). To sort by a transform of the column instead (e.g.
87    /// `bucket[N]`, `year`, `truncate[W]`), use [`Self::desc_with_transform`].
88    pub fn desc(self, name: &str, null_order: NullOrder) -> Self {
89        self.desc_with_transform(name, Transform::Identity, null_order)
90    }
91
92    /// Adds a field for sorting in ascending order by a transform of the column's value
93    /// (e.g. `Transform::Bucket(16)`, `Transform::Year`, `Transform::Truncate(4)`).
94    ///
95    /// Whether the transform is valid for the column's type is checked at commit time,
96    /// once the table schema is available (mirroring Java's `SortOrder.Builder.build()`).
97    /// `Transform::Unknown` and `Transform::Void` are rejected at commit time with
98    /// [`ErrorKind::DataInvalid`]. Rejecting `Void` is an intentional authoring policy
99    /// for this action: an all-null sort key adds no ordering information.
100    ///
101    /// Note: `Term` is currently a plain column reference. Once it becomes
102    /// transform-carrying (#2665), sort-order declaration is expected to converge on
103    /// Term-based `asc`/`desc` (as in Java's `SortOrderBuilder`), at which point the
104    /// `_with_transform` variants can be deprecated in its favor.
105    pub fn asc_with_transform(
106        self,
107        name: &str,
108        transform: Transform,
109        null_order: NullOrder,
110    ) -> Self {
111        self.add_sort_field(name, transform, SortDirection::Ascending, null_order)
112    }
113
114    /// Adds a field for sorting in descending order by a transform of the column's value.
115    ///
116    /// See [`Self::asc_with_transform`] for validation and future API direction.
117    pub fn desc_with_transform(
118        self,
119        name: &str,
120        transform: Transform,
121        null_order: NullOrder,
122    ) -> Self {
123        self.add_sort_field(name, transform, SortDirection::Descending, null_order)
124    }
125
126    fn add_sort_field(
127        mut self,
128        name: &str,
129        transform: Transform,
130        sort_direction: SortDirection,
131        null_order: NullOrder,
132    ) -> Self {
133        self.pending_sort_fields.push(PendingSortField {
134            name: name.to_string(),
135            transform,
136            direction: sort_direction,
137            null_order,
138        });
139
140        self
141    }
142}
143
144#[async_trait]
145impl TransactionAction for ReplaceSortOrderAction {
146    async fn commit(self: Arc<Self>, table: &Table) -> Result<ActionCommit> {
147        let current_schema = table.metadata().current_schema();
148        let sort_fields: Result<Vec<SortField>> = self
149            .pending_sort_fields
150            .iter()
151            .map(|p| p.to_sort_field(current_schema))
152            .collect();
153
154        let bound_sort_order = SortOrder::builder()
155            .with_fields(sort_fields?)
156            .build(current_schema)?;
157
158        let updates = vec![
159            TableUpdate::AddSortOrder {
160                sort_order: bound_sort_order,
161            },
162            TableUpdate::SetDefaultSortOrder { sort_order_id: -1 },
163        ];
164
165        let requirements = vec![
166            TableRequirement::CurrentSchemaIdMatch {
167                current_schema_id: current_schema.schema_id(),
168            },
169            TableRequirement::DefaultSortOrderIdMatch {
170                default_sort_order_id: table.metadata().default_sort_order().order_id,
171            },
172        ];
173
174        Ok(ActionCommit::new(updates, requirements))
175    }
176}

314    #[tokio::test]
315    async fn test_sort_order_transform_survives_metadata_json_round_trip() {
316        // Commit a transform-based sort order through a real catalog: the memory
317        // catalog serializes the updated table metadata to a metadata.json file
318        // (`TableMetadata::write_to`), and `load_table` reads that file back and
319        // parses it (`TableMetadata::read_from`). This exercises the full
320        // JSON round-trip of the transform (e.g. `"bucket[16]"`), not just the
321        // in-memory `TableUpdate`.
322        let catalog = new_memory_catalog().await;
323        let table = make_v3_minimal_table_in_catalog(&catalog).await;
324
325        let tx = Transaction::new(&table);
326        let tx = tx
327            .replace_sort_order()
328            .asc_with_transform("x", Transform::Bucket(16), NullOrder::First)
329            .desc_with_transform("y", Transform::Truncate(4), NullOrder::Last)
330            .apply(tx)
331            .unwrap();
332        let committed = tx.commit(&catalog).await.unwrap();
333
334        // Reload from the catalog: this parses the metadata.json written above.
335        let reloaded = catalog.load_table(committed.identifier()).await.unwrap();
336        let sort_order = reloaded.metadata().default_sort_order();
337
338        assert_eq!(sort_order.fields.len(), 2);
339        assert_eq!(sort_order.fields[0].transform, Transform::Bucket(16));
340        assert_eq!(sort_order.fields[0].direction, SortDirection::Ascending);
341        assert_eq!(sort_order.fields[0].null_order, NullOrder::First);
342        assert_eq!(sort_order.fields[1].transform, Transform::Truncate(4));
343        assert_eq!(sort_order.fields[1].direction, SortDirection::Descending);
344        assert_eq!(sort_order.fields[1].null_order, NullOrder::Last);
345    }
