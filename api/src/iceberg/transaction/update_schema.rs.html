33// Default ID for a new column. This will be re-assigned to a fresh ID at commit time.
34const DEFAULT_FIELD_ID: i32 = 0;
35
36/// Declarative specification for adding a column in an [`UpdateSchemaAction`].
37///
38/// Use helper constructors such as [`AddColumn::optional`] and [`AddColumn::required`],
39/// optionally combined with the builder's `parent` and `doc` setters via
40/// [`AddColumn::builder`], then pass the value to [`UpdateSchemaAction::add_column`].
41#[derive(Clone, TypedBuilder)]
42pub struct AddColumn {
43    #[builder(default = None, setter(strip_option, into))]
44    parent: Option<String>,
45    #[builder(setter(into))]
46    name: String,
47    #[builder(default = false)]
48    required: bool,
49    field_type: Type,
50    #[builder(default = None, setter(strip_option, into))]
51    doc: Option<String>,
52    #[builder(default = None, setter(strip_option))]
53    initial_default: Option<Literal>,
54    #[builder(default = None, setter(strip_option))]
55    write_default: Option<Literal>,
56}
57
58impl AddColumn {
59    /// Create a root-level optional column specification.
60    pub fn optional(name: impl ToString, field_type: Type) -> Self {
61        Self::builder()
62            .name(name.to_string())
63            .field_type(field_type)
64            .required(false)
65            .build()
66    }
67
68    /// Create a root-level required column specification.
69    pub fn required(name: impl ToString, field_type: Type, initial_default: Literal) -> Self {
70        Self::builder()
71            .name(name.to_string())
72            .field_type(field_type)
73            .required(true)
74            .initial_default(initial_default.clone())
75            .write_default(initial_default)
76            .build()
77    }
78
79    fn to_nested_field(&self) -> NestedFieldRef {
80        let mut field = NestedField::new(
81            DEFAULT_FIELD_ID,
82            self.name.clone(),
83            self.field_type.clone(),
84            self.required,
85        );
86
87        field.doc = self.doc.clone();
88        field.initial_default = self.initial_default.clone();
89        field.write_default = self.write_default.clone();
90        Arc::new(field)
91    }
92}

94/// Schema evolution API modeled after the Java `SchemaUpdate` implementation.
95///
96/// This action accumulates schema modifications (column additions and deletions)
97/// via builder methods. At commit time, it validates all operations against the
98/// current table schema, auto-assigns field IDs from `table.metadata().last_column_id()`,
99/// builds a new schema, and emits `AddSchema` + `SetCurrentSchema` updates with a
100/// `CurrentSchemaIdMatch` requirement.
101///
102/// # Example
103///
104/// ```ignore
105/// let tx = Transaction::new(&table);
106/// let action = tx.update_schema()
107///     .add_column(AddColumn::optional("new_col", Type::Primitive(PrimitiveType::Int)))
108///     .add_column(
109///         AddColumn::optional("email", Type::Primitive(PrimitiveType::String))
110///             .with_parent("person")
111///     )
112///     .delete_column("old_col");
113/// let tx = action.apply(tx).unwrap();
114/// let table = tx.commit(&catalog).await.unwrap();
115/// ```
116#[derive(Clone)]
117pub struct UpdateSchemaAction {
118    additions: Vec<AddColumn>,
119    deletes: Vec<String>,
120}
121
122impl UpdateSchemaAction {
123    /// Creates a new empty `UpdateSchemaAction`.
124    pub(crate) fn new() -> Self {
125        Self {
126            additions: Vec::new(),
127            deletes: Vec::new(),
128        }
129    }
130
131    // --- Root-level additions ---
132
133    /// Add a column to the table schema.
134    ///
135    /// To add a root-level column, leave `AddColumn::parent` as `None`.
136    /// For nested additions, set a parent path.
137    /// If the parent resolves to a map/list, the column is added to map value/list element.
138    pub fn add_column(mut self, add_column: AddColumn) -> Self {
139        self.additions.push(add_column);
140        self
141    }
142
143    // --- Other builder methods ---
144
145    /// Record a column deletion by name.
146    ///
147    /// At commit time, the column must exist in the current schema.
148    pub fn delete_column(mut self, name: impl ToString) -> Self {
149        self.deletes.push(name.to_string());
150        self
151    }
152}

158/// Recursively assign fresh field IDs to a `NestedField` and all its nested sub-fields.
159///
160/// This follows the same recursive pattern as `ReassignFieldIds::reassign_ids_visit_type`
161/// from `crate::spec::schema::id_reassigner`, but operates on new fields with placeholder
162/// IDs rather than reassigning an existing schema. `ReassignFieldIds` cannot be used
163/// directly here because it rejects duplicate old IDs (all new fields share placeholder
164/// ID `DEFAULT_FIELD_ID`).
165fn assign_fresh_ids(field: &NestedField, next_id: &mut i32) -> NestedFieldRef {
166    *next_id += 1;
167    let new_id = *next_id;
168    let new_type = assign_fresh_ids_to_type(&field.field_type, next_id);
169
170    Arc::new(NestedField {
171        id: new_id,
172        name: field.name.clone(),
173        required: field.required,
174        field_type: Box::new(new_type),
175        doc: field.doc.clone(),
176        initial_default: field.initial_default.clone(),
177        write_default: field.write_default.clone(),
178    })
179}
180
181/// Recursively assign fresh field IDs to all nested fields within a `Type`.
182fn assign_fresh_ids_to_type(field_type: &Type, next_id: &mut i32) -> Type {
183    match field_type {
184        Type::Primitive(_) => field_type.clone(),
185        // Variant carries no nested fields, so there is nothing to reassign
186        // (matches id_reassigner.rs).
187        Type::Variant(v) => Type::Variant(*v),
188        Type::Struct(struct_type) => {
189            let new_fields: Vec<NestedFieldRef> = struct_type
190                .fields()
191                .iter()
192                .map(|f| assign_fresh_ids(f, next_id))
193                .collect();
194            Type::Struct(StructType::new(new_fields))
195        }
196        Type::List(list_type) => {
197            let new_element = assign_fresh_ids(&list_type.element_field, next_id);
198            Type::List(ListType {
199                element_field: new_element,
200            })
201        }
202        Type::Map(map_type) => {
203            let new_key = assign_fresh_ids(&map_type.key_field, next_id);
204            let new_value = assign_fresh_ids(&map_type.value_field, next_id);
205            Type::Map(MapType {
206                key_field: new_key,
207                value_field: new_value,
208            })
209        }
210    }
211}

217/// Resolve a parent path to the target struct's parent field ID and a reference
218/// to its `StructType`.
219///
220/// If the parent is a map, navigates to the value field. If a list, navigates to
221/// the element field. The final target must be a struct type.
222fn resolve_parent_target<'a>(
223    base_schema: &'a Schema,
224    parent: &str,
225) -> Result<(i32, &'a StructType)> {
226    base_schema
227        .field_by_name(parent)
228        .ok_or_else(|| {
229            Error::new(
230                ErrorKind::PreconditionFailed,
231                format!("Cannot add column: parent '{parent}' not found"),
232            )
233        })
234        .and_then(|parent_field| match parent_field.field_type.as_ref() {
235            Type::Struct(s) => Ok((parent_field.id, s)),
236            Type::Map(m) => match m.value_field.field_type.as_ref() {
237                Type::Struct(s) => Ok((m.value_field.id, s)),
238                _ => Err(Error::new(
239                    ErrorKind::PreconditionFailed,
240                    format!("Cannot add column: map value of '{parent}' is not a struct"),
241                )),
242            },
243            Type::List(l) => match l.element_field.field_type.as_ref() {
244                Type::Struct(s) => Ok((l.element_field.id, s)),
245                _ => Err(Error::new(
246                    ErrorKind::PreconditionFailed,
247                    format!("Cannot add column: list element of '{parent}' is not a struct"),
248                )),
249            },
250            _ => Err(Error::new(
251                ErrorKind::PreconditionFailed,
252                format!("Cannot add column: parent '{parent}' is not a struct, map, or list"),
253            )),
254        })
255}

261/// Rebuild a slice of fields, applying deletions and additions at every level,
262/// plus any additions keyed by `parent_id` (`None` represents the table root).
263fn rebuild_fields(
264    fields: &[NestedFieldRef],
265    adds: &HashMap<Option<i32>, Vec<NestedFieldRef>>,
266    delete_ids: &HashSet<i32>,
267    parent_id: Option<i32>,
268) -> Vec<NestedFieldRef> {
269    fields
270        .iter()
271        .filter(|f| !delete_ids.contains(&f.id))
272        .map(|f| rebuild_field(f, adds, delete_ids))
273        .chain(adds.get(&parent_id).into_iter().flatten().cloned())
274        .collect()
275}
276
277/// Recursively rebuild a single field. If the field (or any descendant) is a struct
278/// that has pending additions, those additions are appended to the struct's fields.
279/// Fields whose IDs appear in `delete_ids` are filtered out at every struct level.
280fn rebuild_field(
281    field: &NestedFieldRef,
282    adds: &HashMap<Option<i32>, Vec<NestedFieldRef>>,
283    delete_ids: &HashSet<i32>,
284) -> NestedFieldRef {
285    match field.field_type.as_ref() {
286        Type::Primitive(_) | Type::Variant(_) => field.clone(),
287        Type::Struct(s) => {
288            let new_fields = rebuild_fields(s.fields(), adds, delete_ids, Some(field.id));
289            Arc::new(NestedField {
290                id: field.id,
291                name: field.name.clone(),
292                required: field.required,
293                field_type: Box::new(Type::Struct(StructType::new(new_fields))),
294                doc: field.doc.clone(),
295                initial_default: field.initial_default.clone(),
296                write_default: field.write_default.clone(),
297            })
298        }
299        Type::List(l) => {
300            let new_element = rebuild_field(&l.element_field, adds, delete_ids);
301            Arc::new(NestedField {
302                id: field.id,
303                name: field.name.clone(),
304                required: field.required,
305                field_type: Box::new(Type::List(ListType {
306                    element_field: new_element,
307                })),
308                doc: field.doc.clone(),
309                initial_default: field.initial_default.clone(),
310                write_default: field.write_default.clone(),
311            })
312        }
313        Type::Map(m) => {
314            let new_key = rebuild_field(&m.key_field, adds, delete_ids);
315            let new_value = rebuild_field(&m.value_field, adds, delete_ids);
316            Arc::new(NestedField {
317                id: field.id,
318                name: field.name.clone(),
319                required: field.required,
320                field_type: Box::new(Type::Map(MapType {
321                    key_field: new_key,
322                    value_field: new_value,
323                })),
324                doc: field.doc.clone(),
325                initial_default: field.initial_default.clone(),
326                write_default: field.write_default.clone(),
327            })
328        }
329    }
330}

336/// Add entries for added columns to a name mapping, as Java's `MappingUtil.update` does.
337///
338/// Each added field gets an entry, with entries for its nested fields, appended under its
339/// parent's entry. An existing entry in the same struct loses the added name, so a dropped
340/// column no longer claims it. Entries of deleted columns are kept.
341fn add_columns_to_name_mapping(
342    mapping: &NameMapping,
343    adds: &HashMap<Option<i32>, Vec<NestedFieldRef>>,
344) -> NameMapping {
345    let root_adds = adds.get(&None).map_or(&[][..], Vec::as_slice);
346    NameMapping::new(add_columns_to_mapped_fields(
347        mapping.fields().iter(),
348        root_adds,
349        adds,
350    ))
351}
352
353fn add_columns_to_mapped_fields<'a>(
354    fields: impl Iterator<Item = &'a MappedField>,
355    added: &[NestedFieldRef],
356    adds: &HashMap<Option<i32>, Vec<NestedFieldRef>>,
357) -> Vec<MappedField> {
358    let added_ids: HashMap<&str, i32> = added
359        .iter()
360        .map(|add| (add.name.as_str(), add.id))
361        .collect();
362    fields
363        .map(|field| {
364            let nested_added = field
365                .field_id()
366                .and_then(|id| adds.get(&Some(id)))
367                .map_or(&[][..], Vec::as_slice);
368            let nested = add_columns_to_mapped_fields(
369                field.fields().iter().map(Arc::as_ref),
370                nested_added,
371                adds,
372            );
373            let names = field
374                .names()
375                .iter()
376                .filter(|name| {
377                    added_ids
378                        .get(name.as_str())
379                        .is_none_or(|id| Some(*id) == field.field_id())
380                })
381                .cloned()
382                .collect();
383            MappedField::new(field.field_id(), names, nested)
384        })
385        .chain(added.iter().map(|add| create_mapped_field(add, &add.name)))
386        .collect()
387}
388
389/// Create a mapping entry for a field and its nested fields, as Java's `MappingUtil.create`.
390fn create_mapped_field(field: &NestedField, name: &str) -> MappedField {
391    let nested = match field.field_type.as_ref() {
392        Type::Primitive(_) | Type::Variant(_) => vec![],
393        Type::Struct(s) => s
394            .fields()
395            .iter()
396            .map(|f| create_mapped_field(f, &f.name))
397            .collect(),
398        Type::List(l) => vec![create_mapped_field(&l.element_field, LIST_FIELD_NAME)],
399        Type::Map(m) => vec![
400            create_mapped_field(&m.key_field, MAP_KEY_FIELD_NAME),
401            create_mapped_field(&m.value_field, MAP_VALUE_FIELD_NAME),
402        ],
403    };
404    MappedField::new(Some(field.id), vec![name.to_string()], nested)
405}

411#[async_trait]
412impl StatelessAction for UpdateSchemaAction {
413    async fn commit(&self, table: &Table) -> Result<ActionCommit> {
414        let base_schema = table.metadata().current_schema();
415        let mut last_column_id = table.metadata().last_column_id();
416
417        // --- 1. Validate deletes ---
418        let delete_ids = self
419            .deletes
420            .iter()
421            .map(|name: &String| {
422                base_schema
423                    .field_by_name(name)
424                    .ok_or_else(|| {
425                        Error::new(
426                            ErrorKind::PreconditionFailed,
427                            format!("Cannot delete missing column: {name}"),
428                        )
429                    })
430                    .and_then(|field| {
431                        match base_schema
432                            .identifier_field_ids()
433                            .find(|id| *id == field.id)
434                        {
435                            Some(_) => Err(Error::new(
436                                ErrorKind::PreconditionFailed,
437                                format!("Cannot delete identifier field: {name}"),
438                            )),
439                            None => Ok(field.id),
440                        }
441                    })
442            })
443            .collect::<Result<HashSet<i32>>>()?;
444
445        // --- 2. Resolve parents, validate additions, assign IDs, and group by parent ID ---
446        // We assign IDs inline (before grouping) to preserve the caller's insertion order,
447        // since HashMap iteration order is non-deterministic.
448        let mut additions_by_parent: HashMap<Option<i32>, Vec<NestedFieldRef>> = HashMap::new();
449
450        for add in &self.additions {
451            let pending_field = add.to_nested_field();
452
453            // Check that name does not contain `SCHEMA_NAME_DELIMITER`.
454            if pending_field.name.contains(SCHEMA_NAME_DELIMITER) {
455                return Err(Error::new(
456                    ErrorKind::PreconditionFailed,
457                    format!(
458                        "Cannot add column with ambiguous name: {}. Use `AddColumn::with_parent` to add a column to a nested struct.",
459                        pending_field.name
460                    ),
461                ));
462            }
463
464            // Required columns without an initial default need allow_incompatible_changes.
465            if pending_field.required && pending_field.initial_default.is_none() {
466                return Err(Error::new(
467                    ErrorKind::PreconditionFailed,
468                    format!(
469                        "Incompatible change: cannot add required column without an initial default: {}",
470                        pending_field.name
471                    ),
472                ));
473            }
474
475            let parent_id = match &add.parent {
476                None => {
477                    // Root-level: check name conflict against root-level fields.
478                    if let Some(existing) = base_schema.field_by_name(&pending_field.name)
479                        && !delete_ids.contains(&existing.id)
480                    {
481                        return Err(Error::new(
482                            ErrorKind::PreconditionFailed,
483                            format!(
484                                "Cannot add column, name already exists: {}",
485                                pending_field.name
486                            ),
487                        ));
488                    }
489                    None
490                }
491                Some(parent_path) => {
492                    // Nested: resolve parent, check name conflict within parent struct.
493                    let (resolved_parent_id, parent_struct) =
494                        resolve_parent_target(base_schema, parent_path)?;
495
496                    if parent_struct.fields().iter().any(|f| {
497                        f.name == pending_field.name
498                            && !delete_ids.contains(&f.id)
499                            && !delete_ids.contains(&resolved_parent_id)
500                    }) {
501                        return Err(Error::new(
502                            ErrorKind::PreconditionFailed,
503                            format!(
504                                "Cannot add column, name already exists in '{}': {}",
505                                parent_path, pending_field.name
506                            ),
507                        ));
508                    }
509
510                    Some(resolved_parent_id)
511                }
512            };
513
514            // Assign fresh IDs immediately, preserving insertion order.
515            let field = assign_fresh_ids(&pending_field, &mut last_column_id);
516
517            additions_by_parent
518                .entry(parent_id)
519                .or_default()
520                .push(field);
521        }
522
523        // --- 4. Rebuild the schema tree with additions and deletions ---
524        let new_fields = rebuild_fields(
525            base_schema.as_struct().fields(),
526            &additions_by_parent,
527            &delete_ids,
528            None,
529        );
530
531        // --- 5. Build the new schema ---
532        let schema = Schema::builder()
533            .with_fields(new_fields)
534            .with_identifier_field_ids(base_schema.identifier_field_ids())
535            .build()?;
536
537        // Additions under a deleted ancestor are not in the new schema and their IDs can be
538        // reassigned later, so they must not get name mapping entries.
539        additions_by_parent
540            .retain(|parent_id, _| parent_id.is_none_or(|id| schema.field_by_id(id).is_some()));
541
542        let mut updates = vec![
543            TableUpdate::AddSchema { schema },
544            TableUpdate::SetCurrentSchema { schema_id: -1 },
545        ];
546
547        // --- 6. Add the new columns to the name mapping, if the table has one ---
548        if !additions_by_parent.is_empty()
549            && let Some(mapping_json) = table
550                .metadata()
551                .properties()
552                .get(TableProperties::PROPERTY_DEFAULT_NAME_MAPPING)
553        {
554            // Like Java, a mapping that can't be updated is logged and does not fail the update.
555            match mapping_json.parse::<NameMapping>().and_then(|mapping| {
556                serde_json::to_string(&add_columns_to_name_mapping(&mapping, &additions_by_parent))
557                    .map_err(Error::from)
558            }) {
559                Ok(updated_json) => updates.push(TableUpdate::SetProperties {
560                    updates: HashMap::from([(
561                        TableProperties::PROPERTY_DEFAULT_NAME_MAPPING.to_string(),
562                        updated_json,
563                    )]),
564                }),
565                Err(error) => {
566                    tracing::warn!("Failed to update name mapping {mapping_json}: {error}")
567                }
568            }
569        }
570
571        let requirements = vec![TableRequirement::CurrentSchemaIdMatch {
572            current_schema_id: base_schema.schema_id(),
573        }];
574
575        Ok(ActionCommit::new(updates, requirements))
576    }
577}
