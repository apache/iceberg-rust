30/// A trait representing an atomic action that can be part of a transaction.
31///
32/// Implementors of this trait define how a specific action is committed to a table.
33/// Each action is responsible for generating the updates and requirements needed
34/// to modify the table metadata.
35///
36/// An action's intent is immutable once applied to a transaction. Retry-persistent
37/// state lives in the associated [`TransactionAction::State`], which survives replay
38/// attempts of one logical execution but is never shared between executions.
39#[async_trait]
40pub(crate) trait TransactionAction: Clone + Send + Sync + 'static {
41    /// Retry-persistent state exclusively owned by one logical execution of this
42    /// action. Stateless actions implement [`StatelessAction`] instead, which
43    /// supplies `State = ()`.
44    type State: Send + Sync + 'static;
45
46    /// Creates fresh state for one logical execution.
47    ///
48    /// This is infallible and table-independent; table-dependent initialization
49    /// happens during [`TransactionAction::commit`].
50    fn new_state(&self) -> Self::State;
51
52    /// Commits this action against the provided table and returns the resulting updates.
53    /// NOTE: This function is intended for internal use only and should not be called directly by users.
54    ///
55    /// One replay attempt: the action (intent) is borrowed immutably, its execution
56    /// state mutably, and the table reflects the current transaction-local base.
57    ///
58    /// # Arguments
59    ///
60    /// * `state` - The retry-persistent state for this logical execution.
61    /// * `table` - The current state of the table this action should apply to.
62    ///
63    /// # Returns
64    ///
65    /// An `ActionCommit` containing table updates and table requirements,
66    /// or an error if the commit fails.
67    async fn commit(&self, state: &mut Self::State, table: &Table) -> Result<ActionCommit>;
68
69    /// Best-effort cleanup after the transaction reaches a terminal result.
70    ///
71    /// Consumes the action and its state: ownership makes any further use
72    /// of this action after cleanup a compile error.
73    /// Cleanup must not change the already-determined transaction result.
74    ///
75    /// See [`CommitStatus`] for what each terminal status allows cleanup to do.
76    // TODO: invoke this from `Transaction::commit` once terminal status
77    // classification is wired up (stateful transaction RFC, milestone 3).
78    #[allow(dead_code)]
79    async fn cleanup(self, state: Self::State, table: &Table, status: CommitStatus);
80}

82/// Convenience trait for actions that carry no per-attempt state.
83///
84/// Implementing `StatelessAction` yields a [`TransactionAction`] impl with
85/// `State = ()`, a no-op [`TransactionAction::new_state`], a no-op
86/// [`TransactionAction::cleanup`], and a [`TransactionAction::commit`] that
87/// forwards to [`StatelessAction::commit`]. Actions that need retry-persistent
88/// state (e.g. a persistent snapshot producer) implement `TransactionAction`
89/// directly instead.
90#[async_trait]
91pub(crate) trait StatelessAction: Clone + Send + Sync + 'static {
92    /// One commit attempt against the current transaction-local table.
93    ///
94    /// Same contract as [`TransactionAction::commit`], minus the state parameter.
95    async fn commit(&self, table: &Table) -> Result<ActionCommit>;
96}
97
98#[async_trait]
99impl<T: StatelessAction> TransactionAction for T {
100    type State = ();
101
102    fn new_state(&self) {}
103
104    async fn commit(&self, _state: &mut (), table: &Table) -> Result<ActionCommit> {
105        StatelessAction::commit(self, table).await
106    }
107
108    async fn cleanup(self, _state: (), _table: &Table, _status: CommitStatus) {}
109}

111/// Classification of a transaction's terminal result, consumed by
112/// [`TransactionAction::cleanup`] to determine what is safe to delete.
113///
114/// The transaction/catalog layer determines this classification; actions
115/// consume it rather than independently interpreting catalog errors.
116// TODO: produce this classification in `Transaction::commit` once terminal
117// cleanup is wired up (stateful transaction RFC, milestone 3).
118#[allow(dead_code)]
119#[derive(Debug, Clone, Copy, PartialEq, Eq)]
120pub(crate) enum CommitStatus {
121    /// The catalog confirmed that the commit was applied. Cleanup may remove
122    /// owned artifacts not retained by the committed result.
123    Committed,
124    /// The transaction definitively did not commit and will not retry.
125    /// Owned artifacts are deletable.
126    Failed,
127    /// The commit request was submitted, but its outcome could not be
128    /// resolved: the catalog may or may not have applied it. Cleanup must
129    /// delete nothing, and the transaction must not be retried.
130    ///
131    /// Example: every action executed and validated successfully, but the
132    /// connection failed while awaiting the catalog's response to
133    /// `update_table`.
134    Unknown,
135}

137/// An entry in a transaction, pairing an action's immutable intent with the
138/// retry-persistent state of one logical execution.
139///
140/// The pairing is preserved by construction: the entry is created with fresh
141/// state and owns both exclusively, so terminal cleanup can consume them together.
142pub(crate) struct ActionEntry<A: TransactionAction> {
143    action: A,
144    state: A::State,
145}
146
147impl<A: TransactionAction> ActionEntry<A> {
148    fn new(action: A) -> Self {
149        let state = action.new_state();
150        Self { action, state }
151    }
152
153    /// The action (intent) held by this entry.
154    #[allow(dead_code)]
155    pub(crate) fn action(&self) -> &A {
156        &self.action
157    }
158}
159
160/// Object-safe adapter over [`ActionEntry`], allowing a transaction to store
161/// heterogeneous entries while keeping each action paired with its own state.
162#[async_trait]
163pub(crate) trait DynActionEntry: AsAny + Send + Sync {
164    /// One replay attempt against the current transaction-local table.
165    async fn commit(&mut self, table: &Table) -> Result<ActionCommit>;
166
167    /// Best-effort terminal cleanup, consuming the entry.
168    // TODO: invoke from `Transaction::commit` once terminal status
169    // classification is wired up (stateful transaction RFC, milestone 3).
170    #[allow(dead_code)]
171    async fn cleanup(self: Box<Self>, table: &Table, status: CommitStatus);
172
173    /// Creates a new entry for a new logical execution of the same action:
174    /// the action (intent) is cloned, and it is paired with fresh state from
175    /// [`TransactionAction::new_state`]. Accrued retry state is never carried
176    /// over to the new entry.
177    #[allow(dead_code)]
178    fn clone_with_new_state(&self) -> TransactionActionEntry;
179}
180
181#[async_trait]
182impl<A: TransactionAction> DynActionEntry for ActionEntry<A> {
183    async fn commit(&mut self, table: &Table) -> Result<ActionCommit> {
184        self.action.commit(&mut self.state, table).await
185    }
186
187    async fn cleanup(self: Box<Self>, table: &Table, status: CommitStatus) {
188        let ActionEntry { action, state } = *self;
189        action.cleanup(state, table, status).await
190    }
191
192    fn clone_with_new_state(&self) -> TransactionActionEntry {
193        Box::new(ActionEntry::new(self.action.clone()))
194    }
195}

197/// A helper trait for applying a `TransactionAction` to a `Transaction`.
198///
199/// This is implemented for all `TransactionAction` types
200/// to allow easy chaining of actions into a transaction context.
201pub trait ApplyTransactionAction {
202    /// Adds this action to the given transaction.
203    ///
204    /// # Arguments
205    ///
206    /// * `tx` - The transaction to apply the action to.
207    ///
208    /// # Returns
209    ///
210    /// The modified transaction containing this action, or an error if the operation fails.
211    fn apply(self, tx: Transaction) -> Result<Transaction>;
212}
213
214impl<T: TransactionAction> ApplyTransactionAction for T {
215    fn apply(self, mut tx: Transaction) -> Result<Transaction>
216    where Self: Sized {
217        tx.actions.push(Box::new(ActionEntry::new(self)));
218        Ok(tx)
219    }
220}

222/// The result of committing a `TransactionAction`.
223///
224/// This struct contains the updates to apply to the table's metadata
225/// and any preconditions that must be satisfied before the update can be committed.
226pub struct ActionCommit {
227    updates: Vec<TableUpdate>,
228    requirements: Vec<TableRequirement>,
229}
230
231impl ActionCommit {
232    /// Creates a new `ActionCommit` from the given updates and requirements.
233    pub fn new(updates: Vec<TableUpdate>, requirements: Vec<TableRequirement>) -> Self {
234        Self {
235            updates,
236            requirements,
237        }
238    }
239
240    /// Consumes and returns the list of table updates.
241    pub fn take_updates(&mut self) -> Vec<TableUpdate> {
242        take(&mut self.updates)
243    }
244
245    /// Consumes and returns the list of table requirements.
246    pub fn take_requirements(&mut self) -> Vec<TableRequirement> {
247        take(&mut self.requirements)
248    }
249}
