18//! Accumulates position delete rows and renders them as spec-conforming [`RecordBatch`]es.
19//!
20//! [`PositionDeletes`] collects `(file_path, pos)` pairs and emits the two required position
21//! delete columns — `file_path` (`Utf8`) and `pos` (`Int64`) — sorted by `(file_path, pos)`
22//! with duplicate pairs removed, as the spec requires. Rows can be drawn either as one batch
23//! ([`to_record_batch`](PositionDeletes::to_record_batch)) or, to bound peak memory over a
24//! large delete set, lazily as fixed-size batches
25//! ([`to_record_batches`](PositionDeletes::to_record_batches)). Either batch is exactly what
26//! [`PositionDeleteFileWriter`](super::position_delete_writer::PositionDeleteFileWriter)
27//! accepts, so it can be handed straight to that writer.
28//!
29//! Position delete files are a v2 construct. v3 tables use deletion vectors and forbid adding
30//! new position delete files, so a format-version gate must be applied at the
31//! transaction/commit layer before routing v3 writes here.
32
33// Nothing wires `PositionDeletes` into a writer yet; the write-path plumbing lands in a follow-up.
34#![allow(dead_code)]
35

45/// Accumulates `(file_path, pos)` position delete rows.
46///
47/// Keying paths in a [`BTreeMap`] and positions in a [`RoaringTreemap`] deduplicates repeated
48/// `(file_path, pos)` pairs and, when iterated, yields the spec-required ascending
49/// `(file_path, pos)` order for free, so callers may [`insert`](Self::insert) rows in any
50/// order.
51#[derive(Debug, Default)]
52pub(crate) struct PositionDeletes {
53    rows: BTreeMap<String, RoaringTreemap>,
54}
55
56impl PositionDeletes {
57    /// Creates an empty accumulator.
58    pub(crate) fn new() -> Self {
59        Self::default()
60    }
61
62    /// Records that row `pos` of `path` is deleted. `pos` is a row position, so it is
63    /// non-negative by construction.
64    ///
65    /// A duplicate `(file_path, pos)` or a position exceeding `i64::MAX` is a caller bug,
66    /// caught by [`debug_assert!`] in debug builds so it surfaces early in debug/CI (a
67    /// position delete file must not list the same `(file_path, pos)` twice, and the `pos`
68    /// column is `Int64`). In release those assertions compile out and the old behavior
69    /// stands: the backing [`RoaringTreemap`] silently collapses a duplicate pair, and an
70    /// oversized position is still rejected as [`ErrorKind::DataInvalid`] when the batch is
71    /// built.
72    pub(crate) fn insert(&mut self, path: impl Into<String>, pos: u64) {
73        debug_assert!(pos <= i64::MAX as u64, "position {pos} exceeds i64::MAX");
74        let inserted = self.rows.entry(path.into()).or_default().insert(pos);
75        debug_assert!(
76            inserted,
77            "duplicate position delete: the same (file_path, pos) was inserted twice"
78        );
79    }
80
81    /// Returns whether no positions have been recorded. `insert` never leaves an empty
82    /// treemap behind, so an empty map means no positions.
83    pub(crate) fn is_empty(&self) -> bool {
84        self.rows.is_empty()
85    }
86
87    /// Returns the total number of recorded positions across all paths.
88    pub(crate) fn len(&self) -> usize {
89        self.rows
90            .values()
91            .map(|positions| positions.len())
92            .sum::<u64>() as usize
93    }

95    /// Renders all accumulated rows as one spec-conforming position delete [`RecordBatch`].
96    ///
97    /// The batch has exactly the two required position delete columns, built against the
98    /// writer's shared [`position_delete_arrow_schema`](super::position_delete_writer::position_delete_arrow_schema)
99    /// so the reserved field ids stay wired in. Rows are emitted sorted by `(file_path, pos)`
100    /// with duplicate pairs removed. An empty accumulator yields a valid 0-row batch.
101    ///
102    /// This materializes every row at once; prefer [`to_record_batches`](Self::to_record_batches)
103    /// to bound peak memory over a large delete set.
104    pub(crate) fn to_record_batch(&self) -> Result<RecordBatch> {
105        // `usize::MAX` never caps a real position count, so this drains everything into one
106        // batch (a 0-row batch when empty).
107        Self::build_batch(&mut self.stream().peekable(), usize::MAX)
108    }
109
110    /// Lazily renders the accumulated rows as position delete [`RecordBatch`]es of at most
111    /// `max_rows` rows each, so the whole delete set is never materialized at once.
112    ///
113    /// Batches carry the two required columns in ascending `(file_path, pos)` order with
114    /// duplicate pairs removed, split on `max_rows` boundaries; a single path whose positions
115    /// overflow one batch simply resumes in the next. Each batch is built on demand as the
116    /// iterator is polled — nothing is pre-collected — and uses the writer's shared
117    /// [`position_delete_arrow_schema`](super::position_delete_writer::position_delete_arrow_schema).
118    /// An empty accumulator yields no batches.
119    ///
120    /// `max_rows` must be greater than zero. Rather than panic, a `max_rows` of `0` makes the
121    /// iterator yield a single [`ErrorKind::DataInvalid`] error and then stop.
122    pub(crate) fn to_record_batches(
123        &self,
124        max_rows: usize,
125    ) -> impl Iterator<Item = Result<RecordBatch>> + '_ {
126        let mut rows = self.stream().peekable();
127        let mut done = false;
128        std::iter::from_fn(move || {
129            if done {
130                return None;
131            }
132            if max_rows == 0 {
133                // Invalid, but never panic: surface it once and stop the iterator.
134                done = true;
135                return Some(Err(Error::new(
136                    ErrorKind::DataInvalid,
137                    "max_rows must be greater than 0",
138                )));
139            }
140            // Nothing left to emit -> end the iteration.
141            rows.peek()?;
142            let batch = Self::build_batch(&mut rows, max_rows);
143            // A failed batch (e.g. a position overflowing `i64`) leaves its offending row
144            // unconsumed, so stop after the first error rather than retry it forever.
145            if batch.is_err() {
146                done = true;
147            }
148            Some(batch)
149        })
150    }
151
152    /// Flattens the accumulator into its rows in ascending `(file_path, pos)` order.
153    ///
154    /// The `BTreeMap` iterates paths ascending and each `RoaringTreemap` iterates positions
155    /// ascending, so the stream is sorted by `(file_path, pos)` and already deduplicated.
156    fn stream(&self) -> impl Iterator<Item = (&str, u64)> + '_ {
157        self.rows
158            .iter()
159            .flat_map(|(path, positions)| positions.iter().map(move |pos| (path.as_str(), pos)))
160    }
161
162    /// Builds one position delete [`RecordBatch`] of at most `max_rows` rows drained from
163    /// `rows`, which must yield rows already in `(file_path, pos)` order.
164    ///
165    /// Consecutive rows sharing a path form a run appended to the `file_path` column in a
166    /// single [`StringBuilder::append_value_n`] call, keeping that column run-length friendly
167    /// rather than repeating the path per row. The batch is built against the writer's shared
168    /// Arrow projection (a cheap `Arc` clone) so the reserved field ids stay wired in. An
169    /// exhausted `rows` yields a valid 0-row batch.
170    fn build_batch<'a, I>(
171        rows: &mut std::iter::Peekable<I>,
172        max_rows: usize,
173    ) -> Result<RecordBatch>
174    where
175        I: Iterator<Item = (&'a str, u64)>,
176    {
177        let mut path_builder = StringBuilder::new();
178        let mut pos_builder = Int64Builder::new();
179
180        let mut taken = 0usize;
181        while taken < max_rows {
182            let Some(&(path, _)) = rows.peek() else {
183                break;
184            };
185            // Consume this path's run: contiguous rows sharing `path`, up to the batch's cap.
186            let mut run = 0usize;
187            while taken < max_rows {
188                let Some(&(next_path, pos)) = rows.peek() else {
189                    break;
190                };
191                if next_path != path {
192                    break;
193                }
194                // The `pos` column is `Int64`; positions are `u64`, so guard the top of the
195                // range with a checked conversion rather than a silent wrap.
196                let pos = i64::try_from(pos)
197                    .map_err(|_| Error::new(ErrorKind::DataInvalid, "position exceeds i64::MAX"))?;
198                pos_builder.append_value(pos);
199                rows.next();
200                run += 1;
201                taken += 1;
202            }
203            // One append per run of `run` rows, not one per row.
204            path_builder.append_value_n(path, run);
205        }
206
207        let schema = super::position_delete_writer::position_delete_arrow_schema();
208        RecordBatch::try_new(schema, vec![
209            Arc::new(path_builder.finish()) as ArrayRef,
210            Arc::new(pos_builder.finish()) as ArrayRef,
211        ])
212        .map_err(|e| {
213            Error::new(
214                ErrorKind::DataInvalid,
215                format!("Failed to build position delete record batch: {e}"),
216            )
217        })
218    }
219}

246    /// Asserts a batch has the exact two-column position delete schema: names, Arrow types,
247    /// non-null flags, and the reserved field ids in the Parquet field-id metadata.

249        assert_eq!(batch.num_columns(), 2);
250        let schema = batch.schema();
251
252        let path = schema.field(0);
253        assert_eq!(path.name(), RESERVED_COL_NAME_DELETE_FILE_PATH);
254        assert_eq!(path.data_type(), &DataType::Utf8);
255        assert!(!path.is_nullable());
256        assert_eq!(
257            path.metadata().get(PARQUET_FIELD_ID_META_KEY).unwrap(),
258            &RESERVED_FIELD_ID_DELETE_FILE_PATH.to_string()
259        );
260
261        let pos = schema.field(1);
262        assert_eq!(pos.name(), RESERVED_COL_NAME_DELETE_FILE_POS);
263        assert_eq!(pos.data_type(), &DataType::Int64);
264        assert!(!pos.is_nullable());
265        assert_eq!(
266            pos.metadata().get(PARQUET_FIELD_ID_META_KEY).unwrap(),
267            &RESERVED_FIELD_ID_DELETE_FILE_POS.to_string()
268        );
269    }

271    /// Extracts the `(file_path, pos)` columns of a batch as parallel vectors.
272    fn columns(batch: &RecordBatch) -> (Vec<String>, Vec<i64>) {
273        let paths = batch
274            .column(0)
275            .as_any()
276            .downcast_ref::<StringArray>()
277            .unwrap();
278        let positions = batch
279            .column(1)
280            .as_any()
281            .downcast_ref::<Int64Array>()
282            .unwrap();
283        assert_eq!(paths.null_count(), 0);
284        assert_eq!(positions.null_count(), 0);
285        (
286            paths.iter().map(|s| s.unwrap().to_string()).collect(),
287            positions.values().to_vec(),
288        )
289    }

408    /// Drains [`PositionDeletes::to_record_batches`] into per-batch `(paths, positions)`
409    /// column pairs, unwrapping each batch.
410    fn batched_columns(deletes: &PositionDeletes, max_rows: usize) -> Vec<(Vec<String>, Vec<i64>)> {
411        deletes
412            .to_record_batches(max_rows)
413            .map(|batch| columns(&batch.unwrap()))
414            .collect()
415    }

417    /// Concatenates per-batch column pairs into one flattened `(paths, positions)` pair.
418    fn flatten(batches: &[(Vec<String>, Vec<i64>)]) -> (Vec<String>, Vec<i64>) {
419        let mut paths = Vec::new();
420        let mut positions = Vec::new();
421        for (batch_paths, batch_positions) in batches {
422            paths.extend(batch_paths.iter().cloned());
423            positions.extend(batch_positions.iter().copied());
424        }
425        (paths, positions)
426    }

432        for (path, pos) in [

589    /// Wires a real [`PositionDeleteFileWriter`] over a local filesystem for the integration
590    /// test: the file IO, location/name generators, and the Parquet-backed rolling writer
591    /// configured with the [`position_delete_schema`].
592    fn writer_setup(
593        temp_dir: &TempDir,
594    ) -> PositionDeleteFileWriterBuilder<
595        ParquetWriterBuilder,
596        DefaultLocationGenerator,
597        DefaultFileNameGenerator,
598    > {
599        let file_io = FileIO::new_with_fs();
600        let location_gen = DefaultLocationGenerator::with_data_location(
601            temp_dir.path().to_str().unwrap().to_string(),
602        );
603        let file_name_gen =
604            DefaultFileNameGenerator::new("test".to_string(), None, DataFileFormat::Parquet);
605        let parquet_writer_builder = ParquetWriterBuilder::new(
606            WriterProperties::builder().build(),
607            position_delete_schema(),
608        );
609        let rolling_writer_builder = RollingFileWriterBuilder::new_with_default_file_size(
610            parquet_writer_builder,
611            file_io,
612            location_gen,
613            file_name_gen,
614        );
615        PositionDeleteFileWriterBuilder::new(rolling_writer_builder)
616    }

618    #[tokio::test]
619    async fn test_batch_feeds_position_delete_writer() -> Result<()> {
620        let temp_dir = TempDir::new().unwrap();
621        let builder = writer_setup(&temp_dir);
622        let mut writer = builder.build(None).await?;
623
624        let mut deletes = PositionDeletes::new();
625        // Deliberately unsorted; `to_record_batch` sorts to the order the writer needs.
626        deletes.insert("s3://bucket/data/f1.parquet", 2);
627        deletes.insert("s3://bucket/data/f0.parquet", 4);
628        deletes.insert("s3://bucket/data/f0.parquet", 1);
629
630        writer.write(deletes.to_record_batch()?).await?;
631        let data_files = writer.close().await?;
632
633        assert_eq!(data_files.len(), 1);
634        let data_file = &data_files[0];
635        assert_eq!(data_file.content_type(), DataContentType::PositionDeletes);
636        assert_eq!(data_file.record_count, deletes.len() as u64);
637
638        Ok(())
639    }

641    #[tokio::test]
642    async fn test_streamed_batches_feed_position_delete_writer() -> Result<()> {
643        let temp_dir = TempDir::new().unwrap();
644        let builder = writer_setup(&temp_dir);
645        let mut writer = builder.build(None).await?;
646
647        let mut deletes = PositionDeletes::new();
648        // Deliberately unsorted, across paths, more rows than the batch cap so more than one
649        // batch is streamed to the writer.
650        deletes.insert("s3://bucket/data/f1.parquet", 2);
651        deletes.insert("s3://bucket/data/f0.parquet", 4);
652        deletes.insert("s3://bucket/data/f0.parquet", 1);
653        deletes.insert("s3://bucket/data/f1.parquet", 5);
654        deletes.insert("s3://bucket/data/f2.parquet", 9);
655
656        // Each fixed-size batch is a valid position delete batch the writer accepts, and the
657        // batches arrive in the sorted order the writer requires.
658        let mut streamed = 0usize;
659        for batch in deletes.to_record_batches(2) {
660            writer.write(batch?).await?;
661            streamed += 1;
662        }
663        assert_eq!(streamed, 3);
664
665        let data_files = writer.close().await?;
666        assert_eq!(data_files.len(), 1);
667        let data_file = &data_files[0];
668        assert_eq!(data_file.content_type(), DataContentType::PositionDeletes);
669        assert_eq!(data_file.record_count, deletes.len() as u64);
670
671        Ok(())
672    }
