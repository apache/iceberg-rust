18//! This module provides `PositionDeleteFileWriter`.
19//!
20//! A position delete file has two required columns: `file_path` (`string`, field id
21//! [`RESERVED_FIELD_ID_DELETE_FILE_PATH`]) and `pos` (`long`, field id
22//! [`RESERVED_FIELD_ID_DELETE_FILE_POS`]). The spec also allows an optional `row` column
23//! that inlines the deleted row's values; this writer does not support it yet, so batches
24//! must contain only the two required columns. The writer takes batches already shaped as
25//! those two columns (see [`position_delete_schema`]) and sets
26//! [`DataContentType::PositionDeletes`] on the output. It does not sort its input; see
27//! [`PositionDeleteFileWriter::write`].
28//!
29//! Position delete files are a v2 construct. v3 replaces them with deletion vectors and
30//! forbids adding new position delete files, so callers must not route v3 writes here.
31//! This base writer has no format-version gate by design; that gating belongs at the
32//! transaction/commit layer.
33

53/// The canonical Iceberg schema of a position delete file: the required `file_path`
54/// (`string`) and `pos` (`long`) columns with their reserved field ids.
55static POSITION_DELETE_SCHEMA: Lazy<SchemaRef> = Lazy::new(|| {
56    Arc::new(
57        Schema::builder()
58            .with_fields(vec![
59                delete_file_path_field().clone(),
60                delete_file_pos_field().clone(),
61            ])
62            .build()
63            .expect("position delete schema is statically valid"),
64    )
65});
66
67/// [`POSITION_DELETE_SCHEMA`] converted to Arrow, keeping the reserved field ids in each
68/// field's Parquet field-id metadata. Test-only for now: callers configure the writer with
69/// the Iceberg [`position_delete_schema`], so no non-test code needs the Arrow form yet.
70#[cfg(test)]
71static POSITION_DELETE_ARROW_SCHEMA: Lazy<arrow_schema::SchemaRef> = Lazy::new(|| {
72    Arc::new(
73        crate::arrow::schema_to_arrow_schema(&POSITION_DELETE_SCHEMA)
74            .expect("position delete arrow schema is statically valid"),
75    )
76});

93/// Reads a field's Iceberg field id from its Parquet field-id metadata.
94fn field_id(field: &Field) -> Result<i32> {
95    field
96        .metadata()
97        .get(PARQUET_FIELD_ID_META_KEY)
98        .ok_or_else(|| {
99            invalid_data!(
100                "Position delete column `{}` is missing its Iceberg field id metadata.",
101                field.name()
102            )
103        })?
104        .parse::<i32>()
105        .map_err(|e| {
106            invalid_data!(
107                "Position delete column `{}` has an invalid field id: {e}",
108                field.name()
109            )
110        })
111}

113/// Validates that a batch is a position delete file: the `file_path` (`Utf8`) and
114/// `pos` (`Int64`) columns, in order, with the two reserved field ids. Checking it
115/// here gives a clear error before the batch reaches the Parquet writer.
116fn validate_position_delete_batch(batch: &RecordBatch) -> Result<()> {
117    let fields = batch.schema_ref().fields();
118    if fields.len() != 2 {
119        return Err(invalid_data!(
120            "This writer supports only the two required position delete columns (`file_path`, `pos`); \
121             batches with a different column count (e.g. including the optional `row` column) are not supported. Got {} columns.",
122            fields.len()
123        ));
124    }
125
126    let path = &fields[0];
127    let path_id = field_id(path)?;
128    if path_id != RESERVED_FIELD_ID_DELETE_FILE_PATH {
129        return Err(invalid_data!(
130            "The first position delete column must be `file_path` (field id {RESERVED_FIELD_ID_DELETE_FILE_PATH}), but got field id {path_id}."
131        ));
132    }
133    // The canonical schema maps Iceberg `string` to `Utf8` and the file writer is
134    // configured with it, so a `LargeUtf8` column has to be cast to `Utf8` first.
135    if path.data_type() != &DataType::Utf8 {
136        return Err(invalid_data!(
137            "The position delete `file_path` column must be Utf8 (cast it first); got {:?}.",
138            path.data_type()
139        ));
140    }
141    // Required column: a nullable field could write nulls under a required schema.
142    if path.is_nullable() {
143        return Err(invalid_data!(
144            "The position delete `file_path` column must be required (non-nullable)."
145        ));
146    }
147
148    let pos = &fields[1];
149    let pos_id = field_id(pos)?;
150    if pos_id != RESERVED_FIELD_ID_DELETE_FILE_POS {
151        return Err(invalid_data!(
152            "The second position delete column must be `pos` (field id {RESERVED_FIELD_ID_DELETE_FILE_POS}), but got field id {pos_id}."
153        ));
154    }
155    if pos.data_type() != &DataType::Int64 {
156        return Err(invalid_data!(
157            "The position delete `pos` column must be Int64, but got {:?}.",
158            pos.data_type()
159        ));
160    }
161    if pos.is_nullable() {
162        return Err(invalid_data!(
163            "The position delete `pos` column must be required (non-nullable)."
164        ));
165    }
166
167    Ok(())
168}

170/// Builder for [`PositionDeleteFileWriter`].
171#[derive(Debug)]
172pub struct PositionDeleteFileWriterBuilder<
173    B: FileWriterBuilder,
174    L: LocationGenerator,
175    F: FileNameGenerator,
176> {
177    inner: RollingFileWriterBuilder<B, L, F>,
178}
179
180impl<B, L, F> PositionDeleteFileWriterBuilder<B, L, F>
181where
182    B: FileWriterBuilder,
183    L: LocationGenerator,
184    F: FileNameGenerator,
185{
186    /// Create a new `PositionDeleteFileWriterBuilder` using a `RollingFileWriterBuilder`.
187    ///
188    /// The `RollingFileWriterBuilder` must be backed by a file writer configured
189    /// with the [`position_delete_schema`]; the per-batch validation in
190    /// [`PositionDeleteFileWriter::write`] guards against a mismatched batch, but
191    /// the caller is responsible for wiring the same schema into the file writer.
192    pub fn new(inner: RollingFileWriterBuilder<B, L, F>) -> Self {
193        Self { inner }
194    }
195}

207        Ok(PositionDeleteFileWriter {

209            partition_key,
210        })
211    }

214/// Writer used to write position delete files within one spec/partition.
215#[derive(Debug)]
216pub struct PositionDeleteFileWriter<
217    B: FileWriterBuilder,
218    L: LocationGenerator,
219    F: FileNameGenerator,
220> {
221    inner: Option<RollingFileWriter<B, L, F>>,
222    partition_key: Option<PartitionKey>,
223}
224
225#[async_trait::async_trait]
226impl<B, L, F> IcebergWriter for PositionDeleteFileWriter<B, L, F>
227where
228    B: FileWriterBuilder,
229    L: LocationGenerator,
230    F: FileNameGenerator,
231{
232    /// Writes a batch of `(file_path, pos)` records; the shape is validated on every
233    /// call.
234    ///
235    /// The writer does not sort its input. Position delete files must be sorted by
236    /// `file_path` then `pos`, so the caller must supply rows in that order across all
237    /// `write` calls; a sorting writer will remove this requirement.
238    async fn write(&mut self, batch: RecordBatch) -> Result<()> {
239        // Reject a closed writer before validating the batch.
240        let Some(writer) = self.inner.as_mut() else {
241            return Err(Error::new(
242                ErrorKind::Unexpected,
243                "Position delete writer is already closed; cannot write.",
244            ));
245        };
246        validate_position_delete_batch(&batch)?;
247        writer.write(&self.partition_key, &batch).await
248    }
249
250    async fn close(&mut self) -> Result<Vec<DataFile>> {
251        if let Some(writer) = self.inner.take() {
252            writer
253                .close()
254                .await?
255                .into_iter()
256                .map(|mut res| {
257                    res.content(DataContentType::PositionDeletes);
258                    // sort_order_id stays null, as the spec requires for position deletes.
259                    if let Some(pk) = self.partition_key.as_ref() {
260                        res.partition(pk.data().clone());
261                        res.partition_spec_id(pk.spec().spec_id());
262                    }
263                    res.build()
264                        .map_err(|e| invalid_data!("Failed to build position delete file: {e}"))
265                })
266                .collect()
267        } else {
268            Err(Error::new(
269                ErrorKind::Unexpected,
270                "Position delete writer is already closed.",
271            ))
272        }
273    }
274}

384    async fn test_position_delete_writer_round_trip() -> Result<()> {
385        let temp_dir = TempDir::new().unwrap();
386        let (file_io, builder) = writer_setup(&temp_dir);
387        let mut writer = builder.build(None).await?;
388
389        // Sorted by (file_path, pos): f0/1, f0/4, then f1/2.
390        let batch = position_delete_batch(
391            vec![
392                "s3://bucket/data/f0.parquet",
393                "s3://bucket/data/f0.parquet",
394                "s3://bucket/data/f1.parquet",
395            ],
396            vec![1, 4, 2],
397        );
398        writer.write(batch.clone()).await?;
399        let data_files = writer.close().await?;
400
401        assert_eq!(data_files.len(), 1);
402        let data_file = &data_files[0];
403        assert_eq!(data_file.content_type(), DataContentType::PositionDeletes);
404        assert_eq!(data_file.file_format, DataFileFormat::Parquet);
405        assert_eq!(data_file.record_count, 3);
406        // Unpartitioned writer leaves the default (empty) partition / spec id.
407        assert_eq!(data_file.partition, Struct::empty());
408        assert_eq!(data_file.partition_spec_id, 0);
409        // The rolling writer fills in file statistics.
410        assert!(data_file.file_size_in_bytes > 0);
411
412        // The written Parquet file round-trips back to the exact input rows.
413        let read_back = read_back_single(&file_io, data_file, &batch.schema()).await;
414        assert_eq!(read_back, batch);
415
416        Ok(())
417    }

872    async fn read_back_single(
873        file_io: &FileIO,
874        data_file: &DataFile,
875        schema: &arrow_schema::SchemaRef,
876    ) -> RecordBatch {
877        let input_content = file_io
878            .new_input(data_file.file_path.clone())
879            .unwrap()
880            .read()
881            .await
882            .unwrap();
883        let reader = ParquetRecordBatchReaderBuilder::try_new(input_content)
884            .unwrap()
885            .build()
886            .unwrap();
887        let batches = reader.map(|b| b.unwrap()).collect::<Vec<_>>();
888        concat_batches(schema, &batches).unwrap()
889    }
