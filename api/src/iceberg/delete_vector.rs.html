27/// Magic bytes prefixing a serialized `deletion-vector-v1` bitmap, per the Iceberg Puffin spec.
28/// Iceberg-Java stores these as the little-endian int 1681511377 (0x6439D3D1).
29const DV_MAGIC: [u8; 4] = [0xD1, 0xD3, 0x39, 0x64];
30const DV_LENGTH_PREFIX_BYTES: usize = 4;
31const DV_MAGIC_BYTES: usize = 4;
32const DV_CRC_BYTES: usize = 4;
33const DV_MIN_BLOB_BYTES: usize = DV_LENGTH_PREFIX_BYTES + DV_MAGIC_BYTES + DV_CRC_BYTES;
34

37    inner: RoaringTreemap,
38}

57    /// Marks the given `positions` as deleted and returns the number of elements appended.
58    ///
59    /// The input slice must be strictly ordered in ascending order, and every value must be greater than all existing values already in the set.
60    ///
61    /// # Errors
62    ///
63    /// Returns an error if the precondition is not met.
64    pub fn insert_positions(&mut self, positions: &[u64]) -> Result<usize> {
65        if let Err(err) = self.inner.append(positions.iter().copied()) {
66            return Err(Error::new(
67                ErrorKind::PreconditionFailed,
68                "failed to marks rows as deleted".to_string(),
69            )
70            .with_source(err));
71        }
72        Ok(positions.len())
73    }

75    #[allow(unused)]
76    pub fn len(&self) -> u64 {
77        self.inner.len()
78    }

80    /// Parses a `deletion-vector-v1` Puffin blob into a `DeleteVector`.
81    ///
82    /// The layout, defined by the Iceberg Puffin spec and matching Iceberg-Java's
83    /// `BitmapPositionDeleteIndex`, is:
84    ///
85    /// ```text
86    /// [length: u32 big-endian][magic: D1 D3 39 64][vector][crc: u32 big-endian]
87    /// ```
88    ///
89    /// `length` counts the magic and vector bytes (not itself or the CRC). The CRC-32 is
90    /// computed over the magic and vector. `vector` is a roaring bitmap in the portable 64-bit
91    /// format: a directory of 32-bit key / 32-bit roaring bitmap pairs, ordered by unsigned
92    /// comparison of the keys, one bitmap per key.
93    ///
94    /// Cardinality is not checked here. The caller validates the decoded length against the
95    /// delete file's `record_count`, where the manifest metadata is available.
96    ///
97    /// # Errors
98    ///
99    /// Returns [`ErrorKind::DataInvalid`] if the blob is shorter than the minimum, the length
100    /// prefix or CRC does not match, the magic is wrong, the roaring bitmap count exceeds the
101    /// portable format's maximum, the roaring directory's keys are not ordered by unsigned
102    /// comparison, or the roaring payload fails to decode.
103    // Consumed by the scan delete loader once the deletion-vector read path is wired up.
104    #[allow(dead_code)]
105    pub fn deserialize(blob: &[u8]) -> Result<Self> {
106        if blob.len() < DV_MIN_BLOB_BYTES {
107            return Err(Error::new(
108                ErrorKind::DataInvalid,
109                format!(
110                    "deletion-vector-v1 blob is {} bytes, shorter than the {DV_MIN_BLOB_BYTES}-byte minimum",
111                    blob.len()
112                ),
113            ));
114        }
115
116        // The magic and vector, i.e. the bytes covered by both the length prefix and the CRC.
117        let body = &blob[DV_LENGTH_PREFIX_BYTES..blob.len() - DV_CRC_BYTES];
118        verify_length_prefix(&blob[..DV_LENGTH_PREFIX_BYTES], body)?;
119
120        // Verify the CRC before interpreting any bytes so a corrupt blob yields a single clear
121        // error rather than an opaque roaring decode failure.
122        verify_crc(body, &blob[blob.len() - DV_CRC_BYTES..])?;
123
124        let (magic, vector) = body.split_at(DV_MAGIC_BYTES);
125        verify_magic(magic)?;
126
127        let inner = decode_roaring_directory(vector)?;
128
129        Ok(DeleteVector { inner })
130    }
131}
132
133fn verify_length_prefix(mut prefix: &[u8], body: &[u8]) -> Result<()> {
134    let declared_len = prefix.try_get_u32().map_err(|e| {
135        Error::new(
136            ErrorKind::Unexpected,
137            "failed to read the deletion-vector-v1 length prefix",
138        )
139        .with_source(e)
140    })? as usize;
141    if declared_len != body.len() {
142        return Err(Error::new(
143            ErrorKind::DataInvalid,
144            format!(
145                "deletion-vector-v1 length prefix is {declared_len}, expected {}",
146                body.len()
147            ),
148        ));
149    }
150    Ok(())
151}
152
153fn verify_crc(body: &[u8], mut crc_bytes: &[u8]) -> Result<()> {
154    let stored_crc = crc_bytes.try_get_u32().map_err(|e| {
155        Error::new(
156            ErrorKind::Unexpected,
157            "failed to read the deletion-vector-v1 CRC",
158        )
159        .with_source(e)
160    })?;
161    let computed_crc = crc32fast::hash(body);
162    if computed_crc != stored_crc {
163        return Err(Error::new(
164            ErrorKind::DataInvalid,
165            format!(
166                "deletion-vector-v1 CRC mismatch: computed {computed_crc:#010x}, stored {stored_crc:#010x}"
167            ),
168        ));
169    }
170    Ok(())
171}
172
173fn verify_magic(magic: &[u8]) -> Result<()> {
174    if magic != DV_MAGIC {
175        return Err(Error::new(
176            ErrorKind::DataInvalid,
177            format!("deletion-vector-v1 magic mismatch: {magic:02x?}, expected {DV_MAGIC:02x?}"),
178        ));
179    }
180    Ok(())
181}

183// The Puffin spec defines the roaring directory as the bitmaps "ordered by unsigned comparison
184// of the 32-bit keys", with one bitmap per key. Walk it ourselves (rather than
185// `RoaringTreemap::deserialize_from`, which stores keys in a `BTreeMap` via a plain insert and
186// would silently accept a stream with duplicate or out-of-order keys, discarding the earlier
187// bitmap on a duplicate) so a non-conformant blob is rejected instead of decoded into a value
188// that doesn't match what was actually written.
189fn decode_roaring_directory(mut reader: &[u8]) -> Result<RoaringTreemap> {
190    let bitmap_count = reader.try_get_u64_le().map_err(|e| {
191        Error::new(
192            ErrorKind::DataInvalid,
193            "failed to decode deletion-vector-v1 roaring payload",
194        )
195        .with_source(e)
196    })?;
197    // The roaring portable format restricts the bitmap count to [0, 2^32 - 1] (it is stored as a
198    // u64 with the upper 32 bits reserved as zero padding).
199    if bitmap_count > u32::MAX as u64 {
200        return Err(Error::new(
201            ErrorKind::DataInvalid,
202            format!(
203                "deletion-vector-v1 roaring bitmap count {bitmap_count} exceeds the {}-key maximum",
204                u32::MAX
205            ),
206        ));
207    }
208
209    let mut bitmaps = Vec::new();
210    let mut last_key: Option<u32> = None;
211    for _ in 0..bitmap_count {
212        let key = reader.try_get_u32_le().map_err(|e| {
213            Error::new(
214                ErrorKind::DataInvalid,
215                "failed to decode deletion-vector-v1 roaring payload",
216            )
217            .with_source(e)
218        })?;
219        if let Some(last) = last_key
220            && key <= last
221        {
222            return Err(Error::new(
223                ErrorKind::DataInvalid,
224                format!(
225                    "deletion-vector-v1 roaring keys must be ordered by unsigned comparison, got key {key} after {last}"
226                ),
227            ));
228        }
229        last_key = Some(key);
230
231        let bitmap = RoaringBitmap::deserialize_from(&mut reader).map_err(|e| {
232            Error::new(
233                ErrorKind::DataInvalid,
234                "failed to decode deletion-vector-v1 roaring payload",
235            )
236            .with_source(e)
237        })?;
238        bitmaps.push((key, bitmap));
239    }
240
241    // `bitmaps` is already sorted by key, but `roaring` has no constructor that accepts
242    // pre-sorted pairs without re-sorting them; revisit if that changes upstream.
243    Ok(RoaringTreemap::from_bitmaps(bitmaps))
244}

246// Ideally, we'd just wrap `roaring::RoaringTreemap`'s iterator, `roaring::treemap::Iter` here.
247// But right now, it does not have a corresponding implementation of `roaring::bitmap::Iter::advance_to`,
248// which is very handy in ArrowReader::build_deletes_row_selection.
249// There is a PR open on roaring to add this (https://github.com/RoaringBitmap/roaring-rs/pull/314)
250// and if that gets merged then we can simplify `DeleteVectorIterator` here, refactoring `advance_to`
251// to just a wrapper around the underlying iterator's method.
252pub struct DeleteVectorIterator<'a> {
253    // NB: `BitMapIter` was only exposed publicly in https://github.com/RoaringBitmap/roaring-rs/pull/316
254    // which is not yet released. As a consequence our Cargo.toml temporarily uses a git reference for
255    // the roaring dependency.
256    outer: BitmapIter<'a>,
257    inner: Option<DeleteVectorIteratorInner<'a>>,
258}
259
260struct DeleteVectorIteratorInner<'a> {
261    high_bits: u32,
262    bitmap_iter: Iter<'a>,
263}
264
265impl Iterator for DeleteVectorIterator<'_> {
266    type Item = u64;
267
268    fn next(&mut self) -> Option<Self::Item> {
269        if let Some(inner) = &mut self.inner
270            && let Some(inner_next) = inner.bitmap_iter.next()
271        {
272            return Some(u64::from(inner.high_bits) << 32 | u64::from(inner_next));
273        }
274
275        if let Some((high_bits, next_bitmap)) = self.outer.next() {
276            self.inner = Some(DeleteVectorIteratorInner {
277                high_bits,
278                bitmap_iter: next_bitmap.iter(),
279            })
280        } else {
281            return None;
282        }
283
284        self.next()
285    }
286}
287
288impl DeleteVectorIterator<'_> {
289    pub fn advance_to(&mut self, pos: u64) {
290        let hi = (pos >> 32) as u32;
291        let lo = pos as u32;
292
293        let Some(ref mut inner) = self.inner else {
294            return;
295        };
296
297        while inner.high_bits < hi {
298            let Some((next_hi, next_bitmap)) = self.outer.next() else {
299                return;
300            };
301
302            *inner = DeleteVectorIteratorInner {
303                high_bits: next_hi,
304                bitmap_iter: next_bitmap.iter(),
305            }
306        }
307
308        inner.bitmap_iter.advance_to(lo);
309    }
310}
311
312impl BitOrAssign for DeleteVector {
313    fn bitor_assign(&mut self, other: Self) {
314        self.inner.bitor_assign(&other.inner);
315    }
316}

375    // Reproduces Iceberg-Java's `deletion-vector-v1` framing so tests can round-trip through
376    // `deserialize` without a Java writer. Cross-implementation golden fixtures produced by
377    // Iceberg-Java are tracked separately; this only checks that our decode matches our encode.
378    fn frame_dv_blob(vector: &[u8]) -> Vec<u8> {
379        let body_len = DV_MAGIC_BYTES + vector.len();
380        let mut blob = Vec::with_capacity(DV_LENGTH_PREFIX_BYTES + body_len + DV_CRC_BYTES);
381        blob.extend_from_slice(&(body_len as u32).to_be_bytes());
382        blob.extend_from_slice(&DV_MAGIC);
383        blob.extend_from_slice(vector);
384        let crc = crc32fast::hash(&blob[DV_LENGTH_PREFIX_BYTES..]);
385        blob.extend_from_slice(&crc.to_be_bytes());
386        blob
387    }
388
389    fn encode_dv_blob(dv: &DeleteVector) -> Vec<u8> {
390        let mut vector = Vec::with_capacity(dv.inner.serialized_size());
391        dv.inner.serialize_into(&mut vector).unwrap();
392        frame_dv_blob(&vector)
393    }

404        let mut positions: Vec<u64> = dv.iter().collect();

482    // Crafts a raw roaring treemap directory (bitmap count header + key/bitmap entries) so tests
483    // can exercise key-ordering violations that `dv_of`/`encode_dv_blob` can never produce, since
484    // `DeleteVector::insert` always keeps keys unique and ascending.
485    fn raw_roaring_vector(entries: &[(u32, &[u32])]) -> Vec<u8> {
486        let mut vector = Vec::new();
487        vector.extend_from_slice(&(entries.len() as u64).to_le_bytes());
488        for (key, positions) in entries {
489            let mut bitmap = RoaringBitmap::new();
490            for &pos in *positions {
491                bitmap.insert(pos);
492            }
493            vector.extend_from_slice(&key.to_le_bytes());
494            bitmap.serialize_into(&mut vector).unwrap();
495        }
496        vector
497    }
