31/// Index of delete files
32#[derive(Debug, Clone)]
33pub(crate) struct DeleteFileIndex {
34    state: Arc<RwLock<DeleteFileIndexState>>,
35}
36
37#[derive(Debug)]
38enum DeleteFileIndexState {
39    Populating(Arc<Notify>),
40    Populated(PopulatedDeleteFileIndex),
41}
42
43#[derive(Debug)]
44struct PopulatedDeleteFileIndex {
45    global_equality_deletes: Vec<Arc<DeleteFileContext>>,
46    eq_deletes_by_partition: HashMap<Struct, Vec<Arc<DeleteFileContext>>>,
47    pos_deletes_by_partition: HashMap<Struct, Vec<Arc<DeleteFileContext>>>,
48    pos_deletes_by_path: HashMap<String, Vec<Arc<DeleteFileContext>>>,
49    // TODO: Deletion Vector support
50}
51
52/// Determines the single data file referenced by a position delete file, if any.
53///
54/// Checks the explicit `referenced_data_file` field first. Falls back to the
55/// lower and upper bounds of the delete file's `file_path` column: when both
56/// bounds are present and equal, the delete file references exactly one data
57/// file. Unequal or missing bounds mean the delete file may reference multiple
58/// data files, so no single path can be inferred. A bound that is not a string
59/// literal is treated the same way.
60fn referenced_data_file(data_file: &DataFile) -> Option<String> {
61    if let Some(path) = data_file.referenced_data_file() {
62        return Some(path);
63    }
64
65    let lower_bound = data_file
66        .lower_bounds()
67        .get(&RESERVED_FIELD_ID_DELETE_FILE_PATH)?;
68    let upper_bound = data_file
69        .upper_bounds()
70        .get(&RESERVED_FIELD_ID_DELETE_FILE_PATH)?;
71
72    if lower_bound != upper_bound {
73        return None;
74    }
75
76    match lower_bound.literal() {
77        PrimitiveLiteral::String(path) => Some(path.clone()),
78        _ => None,
79    }
80}

82impl DeleteFileIndex {
83    /// create a new `DeleteFileIndex` along with the sender that populates it with delete files
84    pub(crate) fn new(runtime: Runtime) -> (DeleteFileIndex, Sender<DeleteFileContext>) {
85        // TODO: what should the channel limit be?
86        let (tx, rx) = channel(10);
87        let notify = Arc::new(Notify::new());
88        let state = Arc::new(RwLock::new(DeleteFileIndexState::Populating(
89            notify.clone(),
90        )));
91        let delete_file_stream = rx.boxed();
92
93        runtime.io().spawn({
94            let state = state.clone();
95            async move {
96                let delete_files: Vec<DeleteFileContext> =
97                    delete_file_stream.collect::<Vec<_>>().await;
98
99                let populated_delete_file_index = PopulatedDeleteFileIndex::new(delete_files);
100
101                {
102                    let mut guard = state.write().unwrap();
103                    *guard = DeleteFileIndexState::Populated(populated_delete_file_index);
104                }
105                notify.notify_waiters();
106            }
107        });
108
109        (DeleteFileIndex { state }, tx)
110    }
111
112    /// Gets all the delete files that apply to the specified data file.
113    pub(crate) async fn get_deletes_for_data_file(
114        &self,
115        data_file: &DataFile,
116        seq_num: Option<i64>,
117    ) -> Vec<FileScanTaskDeleteFile> {
118        // Create the `Notified` while holding the read lock. The read lock ensures that
119        // when we go inside it, either the state is already at Populated or it is still
120        // at Populating AND `notify_waiters()` has not been called yet. Any `Notified`
121        // created before the invocation of `notify_waiters()` will be notified by it
122        // even if `await` has not been called on it yet.
123        let notified = {
124            let guard = self.state.read().unwrap();
125            match &*guard {
126                DeleteFileIndexState::Populating(notifier) => notifier.clone().notified_owned(),
127                DeleteFileIndexState::Populated(index) => {
128                    return index.get_deletes_for_data_file(data_file, seq_num);
129                }
130            }
131        };
132
133        notified.await;
134
135        let guard = self.state.read().unwrap();
136        match guard.deref() {
137            DeleteFileIndexState::Populated(index) => {
138                index.get_deletes_for_data_file(data_file, seq_num)
139            }
140            _ => unreachable!("Cannot be any other state than loaded"),
141        }
142    }
143}

145impl PopulatedDeleteFileIndex {
146    /// Creates a new populated delete file index from a list of delete file contexts, which
147    /// allows for fast lookup when determining which delete files apply to a given data file.
148    ///
149    /// 1. Position deletes that reference a single data file, either through the
150    ///    `referenced_data_file` field or through equal `file_path` column bounds,
151    ///    are indexed by that data file's path.
152    /// 2. All other position deletes are indexed by the partition extracted from
153    ///    their manifest entry.
154    /// 3. Equality deletes stored with an unpartitioned spec are applied as global
155    ///    deletes, per the spec. All other equality deletes are indexed by partition.
156    fn new(files: Vec<DeleteFileContext>) -> PopulatedDeleteFileIndex {
157        let mut eq_deletes_by_partition: HashMap<Struct, Vec<Arc<DeleteFileContext>>> =
158            HashMap::default();
159        let mut pos_deletes_by_partition: HashMap<Struct, Vec<Arc<DeleteFileContext>>> =
160            HashMap::default();
161        let mut pos_deletes_by_path: HashMap<String, Vec<Arc<DeleteFileContext>>> =
162            HashMap::default();
163
164        let mut global_equality_deletes: Vec<Arc<DeleteFileContext>> = vec![];
165
166        files.into_iter().for_each(|ctx| {
167            let arc_ctx = Arc::new(ctx);
168
169            let partition = arc_ctx.manifest_entry.data_file().partition();
170
171            match arc_ctx.manifest_entry.content_type() {
172                DataContentType::PositionDeletes => {
173                    if let Some(path) = referenced_data_file(arc_ctx.manifest_entry.data_file()) {
174                        pos_deletes_by_path.entry(path).or_default().push(arc_ctx);
175                    } else {
176                        pos_deletes_by_partition
177                            .entry(partition.clone())
178                            .or_default()
179                            .push(arc_ctx);
180                    }
181                }
182                DataContentType::EqualityDeletes => {
183                    // The spec states that "Equality delete files stored with an unpartitioned spec are applied as global deletes".
184                    if partition.fields().is_empty() {
185                        global_equality_deletes.push(arc_ctx);
186                    } else {
187                        eq_deletes_by_partition
188                            .entry(partition.clone())
189                            .or_default()
190                            .push(arc_ctx);
191                    }
192                }
193                _ => unreachable!(),
194            }
195        });
196
197        PopulatedDeleteFileIndex {
198            global_equality_deletes,
199            eq_deletes_by_partition,
200            pos_deletes_by_partition,
201            pos_deletes_by_path,
202        }
203    }
204
205    /// Determine all the delete files that apply to the provided `DataFile`.
206    fn get_deletes_for_data_file(
207        &self,
208        data_file: &DataFile,
209        seq_num: Option<i64>,
210    ) -> Vec<FileScanTaskDeleteFile> {
211        let mut results = vec![];
212
213        self.global_equality_deletes
214            .iter()
215            // filter that returns true if the provided delete file's sequence number is **greater than** `seq_num`
216            .filter(|&delete| {
217                seq_num
218                    .map(|seq_num| delete.manifest_entry.sequence_number() > Some(seq_num))
219                    .unwrap_or_else(|| true)
220            })
221            .for_each(|delete| results.push(delete.as_ref().into()));
222
223        if let Some(deletes) = self.eq_deletes_by_partition.get(data_file.partition()) {
224            deletes
225                .iter()
226                // filter that returns true if the provided delete file's sequence number is **greater than** `seq_num`
227                .filter(|&delete| {
228                    seq_num
229                        .map(|seq_num| delete.manifest_entry.sequence_number() > Some(seq_num))
230                        .unwrap_or_else(|| true)
231                        && data_file.partition_spec_id == delete.partition_spec_id
232                })
233                .for_each(|delete| results.push(delete.as_ref().into()));
234        }
235
236        if let Some(deletes) = self.pos_deletes_by_partition.get(data_file.partition()) {
237            deletes
238                .iter()
239                // filter that returns true if the provided delete file's sequence number is **greater than or equal to** `seq_num`
240                .filter(|&delete| {
241                    seq_num
242                        .map(|seq_num| delete.manifest_entry.sequence_number() >= Some(seq_num))
243                        .unwrap_or_else(|| true)
244                        && data_file.partition_spec_id == delete.partition_spec_id
245                })
246                .for_each(|delete| results.push(delete.as_ref().into()));
247        }
248
249        // Position deletes indexed by the exact path of the data file they reference.
250        // An exact path match is sufficient proof that the delete applies, so no
251        // partition spec id check is performed.
252        if let Some(deletes) = self.pos_deletes_by_path.get(data_file.file_path()) {
253            deletes
254                .iter()
255                // filter that returns true if the provided delete file's sequence number is **greater than or equal to** `seq_num`
256                .filter(|&delete| {
257                    seq_num
258                        .map(|seq_num| delete.manifest_entry.sequence_number() >= Some(seq_num))
259                        .unwrap_or(true)
260                })
261                .for_each(|delete| results.push(delete.as_ref().into()));
262        }
263
264        results
265    }
266}

665        let partition = Struct::from_iter([Some(Literal::long(100))]);
666        let other_partition = Struct::from_iter([Some(Literal::long(200))]);
667
668        // The data file lives in a different partition than the delete file.
669        let data_file = build_partitioned_data_file(&other_partition, 1);
670        let same_partition_neighbor = build_partitioned_data_file(&partition, 1);
671
672        // A partitioned pos delete with a referenced path must route by path:
673        // it applies to the referenced file in another partition, and not to
674        // a different file in the delete's own partition.
675        let mut pos_delete = build_pos_delete_referencing(data_file.file_path());
676        pos_delete.partition = partition.clone();
677        pos_delete.partition_spec_id = 1;
678

682        }]);
683
684        let deletes_for_referenced = index.get_deletes_for_data_file(&data_file, Some(0));
685        assert_eq!(deletes_for_referenced.len(), 1);
686        assert_eq!(deletes_for_referenced[0].file_path, pos_delete.file_path());
687
688        assert!(
689            index

758    #[test]
759    fn test_deletion_vector_context_carries_coordinates() {
760        // A deletion vector is a PositionDeletes entry stored as a Puffin blob, located by
761        // content_offset / content_size_in_bytes and scoped by referenced_data_file. Those
762        // three fields, and the record count its bitmap is validated against, must survive the
763        // conversion into a FileScanTaskDeleteFile so the loader can find and apply the blob.
764        let dv = DataFileBuilder::default()
765            .file_path("s3://bucket/data/part-0.parquet-deletes.puffin".to_string())
766            .file_format(DataFileFormat::Puffin)
767            .content(DataContentType::PositionDeletes)
768            .record_count(3)
769            .referenced_data_file(Some("s3://bucket/data/part-0.parquet".to_string()))
770            .content_offset(Some(4))
771            .content_size_in_bytes(Some(40))
772            .partition(Struct::empty())
773            .partition_spec_id(0)
774            .file_size_in_bytes(44)
775            .build()
776            .unwrap();
777
778        let ctx = DeleteFileContext {
779            manifest_entry: build_added_manifest_entry(5, &dv).into(),
780            partition_spec_id: 0,
781        };
782
783        let task: FileScanTaskDeleteFile = (&ctx).into();
784        assert_eq!(task.file_type, DataContentType::PositionDeletes);
785        assert_eq!(task.content_offset, Some(4));
786        assert_eq!(task.content_size_in_bytes, Some(40));
787        assert_eq!(task.record_count, Some(3));
788        assert_eq!(
789            task.referenced_data_file.as_deref(),
790            Some("s3://bucket/data/part-0.parquet")
791        );
792    }
