52/// A task to scan part of file.
53#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, TypedBuilder)]
54#[builder(
55    field_defaults(setter(prefix = "with_")),
56    build_method(into = Result<FileScanTask>)
57)]
58pub struct FileScanTask {
59    /// The total size of the data file in bytes, from the manifest entry.
60    /// Used to skip a stat/HEAD request when reading Parquet footers.
61    file_size_in_bytes: u64,
62    /// The start offset of the file to scan.
63    start: u64,
64    /// The length of the file to scan.
65    length: u64,
66    /// The number of records in the file to scan.
67    ///
68    /// This is an optional field, and only available if we are
69    /// reading the entire data file.
70    #[builder(default)]
71    record_count: Option<u64>,
72
73    /// The first row id assigned to the data file.
74    ///
75    /// Used to derive the `_row_id` metadata column: for a row without an
76    /// explicit `_row_id`, it is this value plus the row's ordinal position.
77    #[serde(skip_serializing_if = "Option::is_none")]
78    #[builder(default)]
79    first_row_id: Option<i64>,
80
81    /// The data sequence number of the file, as opposed to its file sequence
82    /// number: the sequence number preserved when a file is carried forward
83    /// across a rewrite. May be null for an existing entry in a malformed
84    /// manifest that lacks one.
85    ///
86    /// Used to derive the `_last_updated_sequence_number` metadata column.
87    #[serde(skip_serializing_if = "Option::is_none")]
88    #[builder(default)]
89    data_sequence_number: Option<i64>,
90
91    /// The data file path corresponding to the task.
92    data_file_path: String,
93
94    /// The format of the file to scan.
95    data_file_format: DataFileFormat,
96
97    /// The schema of the file to scan.
98    schema: SchemaRef,
99    /// The field ids to project.
100    project_field_ids: Vec<i32>,
101    /// The predicate to filter.
102    #[serde(skip_serializing_if = "Option::is_none")]
103    #[builder(default)]
104    predicate: Option<BoundPredicate>,
105
106    /// The list of delete files that may need to be applied to this data file
107    #[builder(default)]
108    deletes: Vec<FileScanTaskDeleteFile>,
109
110    /// Partition data from the manifest entry, used to identify which columns can use
111    /// constant values from partition metadata vs. reading from the data file.
112    /// Per the Iceberg spec, only identity-transformed partition fields should use constants.
113    #[serde(default)]
114    #[serde(skip_serializing_if = "Option::is_none")]
115    #[serde(serialize_with = "serialize_not_implemented")]
116    #[serde(deserialize_with = "deserialize_not_implemented")]
117    #[builder(default)]
118    partition: Option<Struct>,
119
120    /// The partition spec for this file, used to distinguish identity transforms
121    /// (which use partition metadata constants) from non-identity transforms like
122    /// bucket/truncate (which must read source columns from the data file).
123    #[serde(default)]
124    #[serde(skip_serializing_if = "Option::is_none")]
125    #[serde(serialize_with = "serialize_not_implemented")]
126    #[serde(deserialize_with = "deserialize_not_implemented")]
127    #[builder(default)]
128    partition_spec: Option<Arc<PartitionSpec>>,
129
130    /// Name mapping from table metadata (property: schema.name-mapping.default),
131    /// used to resolve field IDs from column names when Parquet files lack field IDs
132    /// or have field ID conflicts.
133    #[serde(default)]
134    #[serde(skip_serializing_if = "Option::is_none")]
135    #[serde(serialize_with = "serialize_not_implemented")]
136    #[serde(deserialize_with = "deserialize_not_implemented")]
137    #[builder(default)]
138    name_mapping: Option<Arc<NameMapping>>,
139
140    /// The unified partition type across all specs in the table.
141    /// When `RESERVED_FIELD_ID_PARTITION` is in the projected field IDs, the reader
142    /// uses this type along with the task's partition_spec and partition data to
143    /// materialize the `_partition` struct column at read time.
144    ///
145    /// This is a table-level value (same for all tasks in a scan), stored per-task
146    /// so that readers are self-contained without needing back-pointers to table
147    /// metadata. The cost is one Arc clone per task.
148    /// Serde: not yet implemented (same pattern as partition, partition_spec, name_mapping).
149    #[serde(default)]
150    #[serde(skip_serializing_if = "Option::is_none")]
151    #[serde(serialize_with = "serialize_not_implemented")]
152    #[serde(deserialize_with = "deserialize_not_implemented")]
153    #[builder(default)]
154    unified_partition_type: Option<Arc<StructType>>,
155
156    /// Whether this scan task should treat column names as case-sensitive when binding predicates.
157    case_sensitive: bool,
158
159    /// Key metadata for encrypted data files (Parquet Modular Encryption).
160    /// When present, the reader uses this to build `FileDecryptionProperties`.
161    ///
162    /// Note on the trust boundary: for the standard encryption scheme this
163    /// carries `StandardKeyMetadata`, whose payload is the *plaintext* DEK.
164    /// Because `FileScanTask` derives `Serialize`, that plaintext DEK is part
165    /// of the serialized scan plan should these tasks ever be serialized and sent
166    /// over the network.
167    #[serde(default)]
168    #[serde(skip_serializing_if = "Option::is_none")]
169    #[builder(default)]
170    key_metadata: Option<Box<[u8]>>,
171}

195    pub fn first_row_id(&self) -> Option<i64> {

209    /// Returns the format of the data file.
210    pub fn data_file_format(&self) -> DataFileFormat {
211        self.data_file_format

220    pub fn schema_ref(&self) -> SchemaRef {

224    /// Returns the project field id of this file scan task.
225    pub fn project_field_ids(&self) -> &[i32] {
226        &self.project_field_ids
227    }

229    /// Returns the predicate of this file scan task.
230    pub fn predicate(&self) -> Option<&BoundPredicate> {
231        self.predicate.as_ref()
232    }

235    pub fn deletes(&self) -> &[FileScanTaskDeleteFile] {
236        &self.deletes

239    /// Returns the partition data from the manifest entry.
240    pub fn partition(&self) -> Option<&Struct> {
241        self.partition.as_ref()
242    }

244    /// Returns the partition spec for the data file.
245    pub fn partition_spec(&self) -> Option<&Arc<PartitionSpec>> {
246        self.partition_spec.as_ref()
247    }
248
249    /// Returns the name mapping used to resolve field ids.
250    pub fn name_mapping(&self) -> Option<&Arc<NameMapping>> {
251        self.name_mapping.as_ref()
252    }
253
254    /// Returns the unified partition type across all table partition specs.
255    pub fn unified_partition_type(&self) -> Option<&Arc<StructType>> {
256        self.unified_partition_type.as_ref()
257    }
258
259    /// Returns whether names are treated as case-sensitive.
260    pub fn case_sensitive(&self) -> bool {
261        self.case_sensitive
262    }
263
264    /// Returns the key metadata for the encrypted data file.
265    pub fn key_metadata(&self) -> Option<&[u8]> {
266        self.key_metadata.as_deref()
267    }
268
269    fn validate(&self) -> Result<()> {
270        match (self.partition.as_ref(), self.partition_spec.as_deref()) {
271            (None, None) => Ok(()),
272            (None, Some(partition_spec)) if partition_spec.is_unpartitioned() => Ok(()),
273            (None, Some(_)) => Err(Error::new(
274                ErrorKind::DataInvalid,
275                "FileScanTask with a partitioned spec requires partition values",
276            )),
277            (Some(partition), None) if partition.fields().is_empty() => Ok(()),
278            (Some(_), None) => Err(Error::new(
279                ErrorKind::DataInvalid,
280                "Non-empty FileScanTask partition requires a partition spec",
281            )),
282            (Some(partition), Some(partition_spec))
283                if partition.fields().len() != partition_spec.fields().len() =>
284            {
285                Err(Error::new(
286                    ErrorKind::DataInvalid,
287                    format!(
288                        "FileScanTask partition has {} fields but partition spec has {} fields",
289                        partition.fields().len(),
290                        partition_spec.fields().len()
291                    ),
292                ))
293            }
294            (Some(_), Some(partition_spec)) => {
295                partition_spec.partition_type(&self.schema)?;
296                Ok(())
297            }
298        }
299    }
300}

302impl From<FileScanTask> for Result<FileScanTask> {
303    fn from(task: FileScanTask) -> Self {
304        task.validate()?;
305        Ok(task)
306    }
307}

309#[derive(Debug)]
310pub(crate) struct DeleteFileContext {
311    pub(crate) manifest_entry: ManifestEntryRef,
312    pub(crate) partition_spec_id: i32,
313}
314
315impl From<&DeleteFileContext> for FileScanTaskDeleteFile {
316    fn from(ctx: &DeleteFileContext) -> Self {
317        FileScanTaskDeleteFile::builder()
318            .with_file_path(ctx.manifest_entry.file_path().to_string())
319            .with_file_size_in_bytes(ctx.manifest_entry.file_size_in_bytes())
320            .with_file_type(ctx.manifest_entry.content_type())
321            .with_partition_spec_id(ctx.partition_spec_id)
322            .with_equality_ids(ctx.manifest_entry.data_file.equality_ids.clone())
323            .with_referenced_data_file(ctx.manifest_entry.data_file.referenced_data_file.clone())
324            .with_content_offset(ctx.manifest_entry.data_file.content_offset)
325            .with_content_size_in_bytes(ctx.manifest_entry.data_file.content_size_in_bytes)
326            .with_record_count(Some(ctx.manifest_entry.record_count()))
327            .with_key_metadata(
328                ctx.manifest_entry
329                    .data_file
330                    .key_metadata
331                    .as_deref()
332                    .map(Box::from),
333            )
334            .build()
335    }
336}

338/// A task to scan part of file.
339#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, TypedBuilder)]
340#[builder(field_defaults(setter(prefix = "with_")))]
341pub struct FileScanTaskDeleteFile {
342    /// The delete file path
343    pub file_path: String,
344
345    /// The total size of the delete file in bytes, from the manifest entry.
346    pub file_size_in_bytes: u64,
347
348    /// delete file type
349    pub file_type: DataContentType,
350
351    /// partition id
352    pub partition_spec_id: i32,
353
354    /// equality ids for equality deletes (null for anything other than equality-deletes)
355    #[builder(default)]
356    pub equality_ids: Option<Vec<i32>>,
357
358    /// For a deletion vector, the location of the data file whose rows it deletes. Required for
359    /// deletion vectors, and may also be set on a position delete file scoped to one data file.
360    #[serde(default)]
361    #[serde(skip_serializing_if = "Option::is_none")]
362    #[builder(default)]
363    pub referenced_data_file: Option<String>,
364
365    /// For a deletion vector, the offset of the blob within its Puffin file. Set only for
366    /// deletion vectors, where it locates the blob for direct access.
367    #[serde(default)]
368    #[serde(skip_serializing_if = "Option::is_none")]
369    #[builder(default)]
370    pub content_offset: Option<i64>,
371
372    /// For a deletion vector, the length in bytes of the blob within its Puffin file.
373    /// Required together with `content_offset`; both are absent for non-DV delete files.
374    #[serde(default)]
375    #[serde(skip_serializing_if = "Option::is_none")]
376    #[builder(default)]
377    pub content_size_in_bytes: Option<i64>,
378
379    /// The number of records in the delete file, from the manifest entry; for a deletion vector,
380    /// the cardinality of its bitmap. `None` only for a task not built from a manifest entry.
381    #[serde(default)]
382    #[serde(skip_serializing_if = "Option::is_none")]
383    #[builder(default)]
384    pub record_count: Option<u64>,
385
386    /// Key metadata for encrypted delete files (Parquet Modular Encryption).
387    /// When present, the reader uses this to build `FileDecryptionProperties`.
388    ///
389    /// Same plaintext-DEK trust boundary as [`FileScanTask::key_metadata`]:
390    /// this is serialized into the scan plan and crosses the planner -> worker
391    /// channel in the clear for the standard encryption scheme.
392    #[serde(default)]
393    #[serde(skip_serializing_if = "Option::is_none")]
394    #[builder(default)]
395    pub key_metadata: Option<Box<[u8]>>,
396}

404    fn build_file_scan_task(
405        schema: SchemaRef,
406        partition: Option<Struct>,
407        partition_spec: Option<Arc<PartitionSpec>>,
408    ) -> Result<FileScanTask> {
409        FileScanTask::builder()
410            .with_file_size_in_bytes(100)
411            .with_start(0)
412            .with_length(100)
413            .with_data_file_path("data_file_path".to_string())
414            .with_data_file_format(DataFileFormat::Parquet)
415            .with_schema(schema)
416            .with_project_field_ids(vec![])
417            .with_partition(partition)
418            .with_partition_spec(partition_spec)
419            .with_case_sensitive(false)
420            .build()
421    }

423    fn schema_and_spec(
424        primitive_type: PrimitiveType,
425        transform: Transform,
426    ) -> (SchemaRef, Arc<PartitionSpec>) {
427        let schema = Arc::new(
428            Schema::builder()
429                .with_fields(vec![Arc::new(NestedField::required(
430                    1,
431                    "x",
432                    Type::Primitive(primitive_type),
433                ))])
434                .build()
435                .unwrap(),
436        );
437        let partition_spec = Arc::new(
438            PartitionSpec::builder(schema.clone())
439                .add_partition_field("x", "x_partition", transform)
440                .unwrap()
441                .build()
442                .unwrap(),
443        );
444        (schema, partition_spec)
445    }
