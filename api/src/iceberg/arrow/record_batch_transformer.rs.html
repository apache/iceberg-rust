61/// Build a map of field ID to constant value (as Datum) for identity-partitioned fields.
62///
63/// Implements Iceberg spec "Column Projection" rule #1: use partition metadata constants
64/// only for identity-transformed fields. Non-identity transforms (bucket, truncate, year, etc.)
65/// store derived values in partition metadata, so source columns must be read from data files.
66///
67/// Example: For `bucket(4, id)`, partition metadata has `id_bucket = 2` (bucket number),
68/// but the actual `id` values (100, 200, 300) are only in the data file.
69///
70/// Matches Java's `PartitionUtil.constantsMap()` which filters `if (field.transform().isIdentity())`.
71///
72/// # References
73/// - Spec: https://iceberg.apache.org/spec/#column-projection
74/// - Java: core/src/main/java/org/apache/iceberg/util/PartitionUtil.java:constantsMap()
75fn constants_map(
76    partition_spec: &PartitionSpec,
77    partition_data: &Struct,
78    schema: &IcebergSchema,
79) -> Result<HashMap<i32, Datum>> {
80    let mut constants = HashMap::new();
81
82    for (pos, field) in partition_spec.fields().iter().enumerate() {
83        // Only identity transforms should use constant values from partition metadata
84        if matches!(field.transform, Transform::Identity) {
85            // The source column may have been dropped from the schema after the spec was
86            // created. It cannot be projected in that case, so no constant is needed.
87            let Some(iceberg_field) = schema.field_by_id(field.source_id) else {
88                continue;
89            };
90
91            // Ensure the field type is primitive
92            let prim_type = match &*iceberg_field.field_type {
93                crate::spec::Type::Primitive(prim_type) => prim_type,
94                _ => {
95                    return Err(Error::new(
96                        ErrorKind::Unexpected,
97                        format!(
98                            "Partition field {} has non-primitive type {:?}",
99                            field.source_id, iceberg_field.field_type
100                        ),
101                    ));
102                }
103            };
104
105            // Get the partition value for this field
106            // Handle both None (null) and Some(Literal::Primitive) cases
107            match &partition_data[pos] {
108                None => {
109                    // Skip null partition values - they will be resolved as null per Iceberg spec rule #4.
110                    // When a partition value is null, we don't add it to the constants map,
111                    // allowing downstream column resolution to handle it correctly.
112                    continue;
113                }
114                Some(Literal::Primitive(value)) => {
115                    // Create a Datum from the primitive type and value
116                    let datum = Datum::new(prim_type.clone(), value.clone());
117                    constants.insert(field.source_id, datum);
118                }
119                Some(literal) => {
120                    return Err(Error::new(
121                        ErrorKind::Unexpected,
122                        format!(
123                            "Partition field {} has non-primitive value: {:?}",
124                            field.source_id, literal
125                        ),
126                    ));
127                }
128            }
129        }
130    }
131
132    Ok(constants)
133}

135/// Indicates how a particular column in a processed RecordBatch should
136/// be sourced.
137#[derive(Debug)]
138pub(crate) enum ColumnSource {
139    // signifies that a column should be passed through unmodified
140    // from the file's RecordBatch
141    PassThrough {
142        source_index: usize,
143    },
144
145    // signifies that a column from the file's RecordBatch has undergone
146    // type promotion so the source column with the given index needs
147    // to be promoted to the specified type
148    Promote {
149        target_type: DataType,
150        source_index: usize,
151    },
152
153    // Signifies that a new column has been inserted before the column
154    // with index `index`. (we choose "before" rather than "after" so
155    // that we can use usize; if we insert after, then we need to
156    // be able to store -1 here to signify that a new
157    // column is to be added at the front of the column list).
158    // If multiple columns need to be inserted at a given
159    // location, they should all be given the same index, as the index
160    // here refers to the original RecordBatch, not the interim state after
161    // a preceding operation.
162    Add {
163        target_type: DataType,
164        value: Option<PrimitiveLiteral>,
165    },
166
167    // A struct column where each child is a constant primitive value.
168    // Used for the _partition metadata column.
169    AddStructConstant {
170        fields: Fields,
171        child_values: Vec<Option<PrimitiveLiteral>>,
172    },
173
174    // A metadata column read from the file and coalesced with a per-file
175    // fallback: the source column's value where non-null, else `fallback`.
176    // Used for `_last_updated_sequence_number` when the file physically carries
177    // the per-row column. The result is cast to `target_type` so it matches the
178    // (run-end-encoded) type the constant and null paths produce for the column.
179    CoalesceLastUpdatedSeq {
180        source_index: usize,
181        fallback: PrimitiveLiteral,
182        target_type: DataType,
183    },
184    // The iceberg spec refers to other permissible schema evolution actions
185    // (see https://iceberg.apache.org/spec/#schema-evolution):
186    // renaming fields, deleting fields and reordering fields.
187    // Renames only affect the schema of the RecordBatch rather than the
188    // columns themselves, so a single updated cached schema can
189    // be re-used and no per-column actions are required.
190    // Deletion and Reorder can be achieved without needing this
191    // post-processing step by using the projection mask.
192}

194#[derive(Debug)]
195enum BatchTransform {
196    // Indicates that no changes need to be performed to the RecordBatches
197    // coming in from the stream and that they can be passed through
198    // unmodified
199    PassThrough,
200
201    Modify {
202        // Every transformed RecordBatch will have the same schema. We create the
203        // target just once and cache it here. Helpfully, Arc<Schema> is needed in
204        // the constructor for RecordBatch, so we don't need an expensive copy
205        // each time we build a new RecordBatch
206        target_schema: Arc<ArrowSchema>,
207
208        // Indicates how each column in the target schema is derived.
209        operations: Vec<ColumnSource>,
210    },
211
212    // Sometimes only the schema will need modifying, for example when
213    // the column names have changed vs the file, but not the column types.
214    // we can avoid a heap allocation per RecordBach in this case by retaining
215    // the existing column Vec.
216    ModifySchema {
217        target_schema: Arc<ArrowSchema>,
218    },
219}

231/// and the `_partition` struct) are stored in a single `constant_fields` map keyed by
232/// field_id. This unified representation (via [`ColumnConstant`]) means the
233/// transformer handles all constant columns through one code path.
234#[derive(Debug)]
235pub(crate) struct RecordBatchTransformerBuilder {
236    snapshot_schema: Arc<IcebergSchema>,
237    projected_iceberg_field_ids: Vec<i32>,
238    constant_fields: HashMap<i32, ColumnConstant>,
239    virtual_fields: HashSet<i32>,
240}

242/// How a metadata (or identity-partition) column's values are supplied.
243///
244/// Covers scalar constants (metadata columns like `_file` and `_spec_id`, and
245/// identity partition source fields), the struct `_partition` column, an all-null
246/// column, and a per-row column coalesced with a per-file fallback.
247#[derive(Debug, Clone, PartialEq)]
248pub(crate) enum ColumnConstant {
249    /// A scalar constant (e.g., `_file` path, `_spec_id`, identity partition values).
250    /// The Datum carries both the Iceberg type and the value.
251    Scalar(Datum),
252    /// A struct constant (the `_partition` column). Each child is a primitive constant
253    /// or null (for partition evolution gaps).
254    Struct(StructConstant),
255    /// An all-null metadata column, carrying its Arrow type. Used when a metadata
256    /// column resolves to null (e.g. `_last_updated_sequence_number` for a file
257    /// with a null `first_row_id`). A distinct variant (rather than `Scalar` with a
258    /// null value) because `Datum` always represents a non-null value; the type is
259    /// computed once so the schema and column-source paths cannot derive it apart.
260    Null(DataType),
261    /// A metadata column physically present in the file that must be coalesced with
262    /// a per-file fallback: read the per-row value, substitute this Datum where the
263    /// per-row value is null (e.g. `_last_updated_sequence_number` on a file that
264    /// carries the column, falling back to the data sequence number).
265    CoalesceLastUpdatedSeq(Datum),
266}

338    /// Add an all-null metadata column for a specific field ID.
339    /// Used for metadata columns that resolve to null (e.g.
340    /// `_last_updated_sequence_number` when the file has a null `first_row_id`).
341    pub(crate) fn with_null_metadata_column(mut self, field_id: i32) -> Result<Self> {
342        let iceberg_field = get_metadata_field(field_id)?;
343        let prim_type = iceberg_field
344            .field_type
345            .as_primitive_type()
346            .ok_or_else(|| {
347                Error::new(
348                    ErrorKind::Unexpected,
349                    format!(
350                        "Metadata column {} has non-primitive type {:?}",
351                        iceberg_field.name, iceberg_field.field_type
352                    ),
353                )
354            })?;
355        // Use the same run-end-encoded type as the non-null constant path, so a
356        // column keeps one Arrow type whether a given file resolves it to a value
357        // or to null.
358        let arrow_type = primitive_type_to_arrow_type_with_ree(prim_type);
359        self.constant_fields
360            .insert(field_id, ColumnConstant::Null(arrow_type));
361        Ok(self)
362    }

412/// Transforms RecordBatches from Parquet files to match the Iceberg table schema.
413///
414/// Handles schema evolution, column reordering, type promotion, and implements the Iceberg spec's
415/// "Column Projection" rules for resolving field IDs "not present" in data files:
416/// 1. Return the value from partition metadata if an Identity Transform exists
417/// 2. Use schema.name-mapping.default metadata to map field id to columns without field id (applied in ArrowReader)
418/// 3. Return the default value if it has a defined initial-default
419/// 4. Return null in all other cases
420///
421/// # Field ID Resolution
422///
423/// Field ID resolution happens in ArrowReader before data is read (matching Java's ReadConf):
424/// - If file has embedded field IDs: trust them (ParquetSchemaUtil.hasIds() = true)
425/// - If file lacks IDs and name_mapping exists: apply name mapping (ParquetSchemaUtil.applyNameMapping())
426/// - If file lacks IDs and no name_mapping: use position-based fallback (ParquetSchemaUtil.addFallbackIds())
427///
428/// By the time RecordBatchTransformer processes data, all field IDs are trustworthy.
429/// This transformer only handles remaining projection rules (#1, #3, #4) for fields still "not present".
430///
431/// # Partition Spec and Data
432///
433/// **Bucket partitioning**: Distinguish identity transforms (use partition metadata constants)
434/// from non-identity transforms like bucket (read from data file) to enable runtime filtering on
435/// bucket-partitioned columns. For example, `bucket(4, id)` stores only the bucket number in
436/// partition metadata, so actual `id` values must be read from the data file.
437///
438/// # References
439/// - Spec: https://iceberg.apache.org/spec/#column-projection
440/// - Java: parquet/src/main/java/org/apache/iceberg/parquet/ReadConf.java (field ID resolution)
441/// - Java: core/src/main/java/org/apache/iceberg/util/PartitionUtil.java (partition constants)
442#[derive(Debug)]
443pub(crate) struct RecordBatchTransformer {
444    snapshot_schema: Arc<IcebergSchema>,
445    projected_iceberg_field_ids: Vec<i32>,
446    // Unified map of all per-file constant columns (metadata, identity partition, _partition struct).
447    constant_fields: HashMap<i32, ColumnConstant>,
448
449    // Field IDs whose data is delivered by the source batch as an arrow-rs virtual column
450    // (e.g. _pos via RowNumber). These fields bypass the snapshot-schema lookup and the
451    // Iceberg projection rules (name mapping / initial-default / null)
452    virtual_fields: HashSet<i32>,
453
454    // BatchTransform gets lazily constructed based on the schema of
455    // the first RecordBatch we receive from the file
456    batch_transform: Option<BatchTransform>,
457}
458
459impl RecordBatchTransformer {
460    pub(crate) fn process_record_batch(
461        &mut self,
462        record_batch: RecordBatch,
463    ) -> Result<RecordBatch> {
464        Ok(match &self.batch_transform {
465            Some(BatchTransform::PassThrough) => record_batch,
466            Some(BatchTransform::Modify {
467                target_schema,
468                operations,
469            }) => {
470                let options = RecordBatchOptions::default()
471                    .with_match_field_names(false)
472                    .with_row_count(Some(record_batch.num_rows()));
473                RecordBatch::try_new_with_options(
474                    Arc::clone(target_schema),
475                    self.transform_columns(record_batch.columns(), operations)?,
476                    &options,
477                )?
478            }
479            Some(BatchTransform::ModifySchema { target_schema }) => {
480                let options = RecordBatchOptions::default()
481                    .with_match_field_names(false)
482                    .with_row_count(Some(record_batch.num_rows()));
483                RecordBatch::try_new_with_options(
484                    Arc::clone(target_schema),
485                    record_batch.columns().to_vec(),
486                    &options,
487                )?
488            }
489            None => {
490                self.batch_transform = Some(Self::generate_batch_transform(
491                    record_batch.schema_ref(),
492                    self.snapshot_schema.as_ref(),
493                    &self.projected_iceberg_field_ids,
494                    &self.constant_fields,
495                    &self.virtual_fields,
496                )?);
497
498                self.process_record_batch(record_batch)?
499            }
500        })
501    }
502
503    // Compare the schema of the incoming RecordBatches to the schema of
504    // the Iceberg snapshot to determine what, if any, transformation
505    // needs to be applied. If the schemas match, we return BatchTransform::PassThrough
506    // to indicate that no changes need to be made. Otherwise, we return a
507    // BatchTransform::Modify containing the target RecordBatch schema and
508    // the list of `ColumnSource`s that indicate how to source each column in
509    // the resulting RecordBatches.
510    fn generate_batch_transform(
511        source_schema: &ArrowSchemaRef,
512        snapshot_schema: &IcebergSchema,
513        projected_iceberg_field_ids: &[i32],
514        constant_fields: &HashMap<i32, ColumnConstant>,
515        virtual_fields: &HashSet<i32>,
516    ) -> Result<BatchTransform> {
517        let mapped_unprojected_arrow_schema = Arc::new(schema_to_arrow_schema(snapshot_schema)?);
518        let field_id_to_mapped_schema_map =
519            Self::build_field_id_to_arrow_schema_map(&mapped_unprojected_arrow_schema)?;
520
521        // Create a new arrow schema by selecting fields from mapped_unprojected,
522        // in the order of the field ids in projected_iceberg_field_ids
523        let fields: Result<Vec<_>> = projected_iceberg_field_ids
524            .iter()
525            .map(|field_id| {
526                match constant_fields.get(field_id) {
527                    Some(ColumnConstant::Struct(pc))
528                        if *field_id == RESERVED_FIELD_ID_PARTITION =>
529                    {
530                        let struct_type = DataType::Struct(pc.fields().clone());
531                        let nullable = pc.fields().is_empty();
532                        let arrow_field = field_with_id(
533                            RESERVED_COL_NAME_PARTITION,
534                            struct_type,
535                            nullable,
536                            *field_id,
537                        );
538                        return Ok(Arc::new(arrow_field));
539                    }
540                    Some(ColumnConstant::Struct(_)) => {
541                        return Err(Error::new(
542                            ErrorKind::Unexpected,
543                            format!(
544                                "Struct column constants are only supported for the `_partition` \
545                                 metadata column (field id {RESERVED_FIELD_ID_PARTITION}), but \
546                                 one was set for field id {field_id}"
547                            ),
548                        ));
549                    }
550                    Some(ColumnConstant::Scalar(datum)) => {
551                        if let Ok(iceberg_field) = get_metadata_field(*field_id) {
552                            let arrow_type = datum_to_arrow_type_with_ree(datum);
553                            let arrow_field = field_with_id(
554                                &iceberg_field.name,
555                                arrow_type,
556                                !iceberg_field.required,
557                                iceberg_field.id,
558                            );
559                            return Ok(Arc::new(arrow_field));
560                        }
561                        // Identity partition constant -- fall through to use the
562                        // mapped schema field (read from file if present).
563                    }
564                    Some(ColumnConstant::Null(arrow_type)) => {
565                        let iceberg_field = get_metadata_field(*field_id)?;
566                        // Always nullable: this arm produces an all-null column regardless
567                        // of the field's declared optionality.
568                        let arrow_field = field_with_id(
569                            &iceberg_field.name,
570                            arrow_type.clone(),
571                            true,
572                            iceberg_field.id,
573                        );
574                        return Ok(Arc::new(arrow_field));
575                    }
576                    Some(ColumnConstant::CoalesceLastUpdatedSeq(datum)) => {
577                        // Same run-end-encoded type as the constant/null paths, so the
578                        // column has one Arrow type whether a file coalesces, uses the
579                        // constant, or is nulled.
580                        let iceberg_field = get_metadata_field(*field_id)?;
581                        let arrow_field = field_with_id(
582                            &iceberg_field.name,
583                            datum_to_arrow_type_with_ree(datum),
584                            true,
585                            iceberg_field.id,
586                        );
587                        return Ok(Arc::new(arrow_field));
588                    }
589                    None => {}
590                }
591
592                if virtual_fields.contains(field_id) {
593                    let virtual_field = get_metadata_field(*field_id)?;
594                    let arrow_type = type_to_arrow_type(&virtual_field.field_type)?;
595                    let arrow_field = field_with_id(
596                        &virtual_field.name,
597                        arrow_type,
598                        !virtual_field.required,
599                        virtual_field.id,
600                    );
601                    return Ok(Arc::new(arrow_field));
602                }
603
604                // Regular fields and identity-partitioned constant fields both exist in the
605                // table schema, so use the mapped Arrow field as-is.
606                Ok(field_id_to_mapped_schema_map
607                    .get(field_id)
608                    .ok_or(Error::new(ErrorKind::Unexpected, "field not found"))?
609                    .0
610                    .clone())
611            })
612            .collect();
613
614        let target_schema = Arc::new(ArrowSchema::new(fields?));
615
616        match Self::compare_schemas(source_schema, &target_schema) {
617            SchemaComparison::Equivalent => Ok(BatchTransform::PassThrough),
618            SchemaComparison::NameChangesOnly => Ok(BatchTransform::ModifySchema { target_schema }),
619            SchemaComparison::Different => Ok(BatchTransform::Modify {
620                operations: Self::generate_transform_operations(
621                    source_schema,
622                    snapshot_schema,
623                    projected_iceberg_field_ids,
624                    field_id_to_mapped_schema_map,
625                    constant_fields,
626                    virtual_fields,
627                )?,
628                target_schema,

633    /// Compares the source and target schemas to decide how much work the
634    /// transform must do. The checks are applied in this order:
635    ///  1. If the schemas have different numbers of fields, they are
636    ///     `Different`: we must rebuild both the schema AND the columns.
637    ///  2. If every field matches positionally in name, data type, and
638    ///     nullability, they are `Equivalent`: the original RecordBatch can be
639    ///     passed through unmodified.
640    ///  3. If they contain the same set of fields but in a different order
641    ///     (detected via set-equality, since field names are unique), they are
642    ///     `Different`: the columns must be rebuilt in the target order.
643    ///  4. Otherwise, the fields align positionally: if any differs in data
644    ///     type or nullability the schemas are `Different` (rebuild schema AND
645    ///     columns); if they differ only in names, it is `NameChangesOnly` and
646    ///     we can relabel the schema BUT keep the original column data.
647    fn compare_schemas(
648        source_schema: &ArrowSchemaRef,
649        target_schema: &ArrowSchemaRef,
650    ) -> SchemaComparison {
651        if source_schema.fields().len() != target_schema.fields().len() {
652            return SchemaComparison::Different;
653        }
654
655        let field_eq = |a: &FieldRef, b: &FieldRef| {
656            a.name() == b.name()
657                && a.data_type() == b.data_type()
658                && a.is_nullable() == b.is_nullable()
659        };
660
661        let positional = || source_schema.fields().iter().zip(target_schema.fields());
662
663        // Field-by-field identical: nothing to do.
664        if positional().all(|(s, t)| field_eq(s, t)) {
665            return SchemaComparison::Equivalent;
666        }
667
668        // Same set of fields but reordered: rebuild in target (selection) order.
669        // Lengths are equal and field names are unique, so this is set-equality.
670        let reordered = target_schema
671            .fields()
672            .iter()
673            .all(|t| source_schema.fields().iter().any(|s| field_eq(s, t)));
674        if reordered {
675            return SchemaComparison::Different;
676        }
677
678        // Not a reordering: any type/nullability difference needs a full
679        // rebuild; otherwise only names differ and we can just relabel.
680        if positional()
681            .any(|(s, t)| s.data_type() != t.data_type() || s.is_nullable() != t.is_nullable())
682        {
683            SchemaComparison::Different
684        } else {
685            SchemaComparison::NameChangesOnly
686        }
687    }

689    fn generate_transform_operations(
690        source_schema: &ArrowSchemaRef,
691        snapshot_schema: &IcebergSchema,
692        projected_iceberg_field_ids: &[i32],
693        field_id_to_mapped_schema_map: HashMap<i32, (FieldRef, usize)>,
694        constant_fields: &HashMap<i32, ColumnConstant>,
695        virtual_fields: &HashSet<i32>,
696    ) -> Result<Vec<ColumnSource>> {
697        let field_id_to_source_schema_map =
698            Self::build_field_id_to_arrow_schema_map(source_schema)?;
699
700        projected_iceberg_field_ids
701            .iter()
702            .map(|field_id| {
703                match constant_fields.get(field_id) {
704                    Some(ColumnConstant::Struct(pc))
705                        if *field_id == RESERVED_FIELD_ID_PARTITION =>
706                    {
707                        return Ok(ColumnSource::AddStructConstant {
708                            fields: pc.fields().clone(),
709                            child_values: pc.child_values().to_vec(),
710                        });
711                    }
712                    Some(ColumnConstant::Struct(_)) => {
713                        return Err(Error::new(
714                            ErrorKind::Unexpected,
715                            format!(
716                                "Struct column constants are only supported for the `_partition` \
717                                 metadata column (field id {RESERVED_FIELD_ID_PARTITION}), but \
718                                 one was set for field id {field_id}"
719                            ),
720                        ));
721                    }
722                    Some(ColumnConstant::Scalar(datum)) => {
723                        // Metadata/virtual fields (like _file, _spec_id) never exist in the
724                        // data file, so they always use their pre-computed constant value.
725                        //
726                        // For identity-partitioned fields, the Iceberg spec's "Column
727                        // Projection" rules only apply to "field ids which are not present in
728                        // a data file". When the column IS present in the Parquet file, it
729                        // must be read from the file; the partition metadata constant is only
730                        // a fallback for when the column is absent (e.g. add_files).
731                        let is_metadata = get_metadata_field(*field_id).is_ok();
732                        let present_in_file =
733                            field_id_to_source_schema_map.contains_key(field_id);
734
735                        if is_metadata || !present_in_file {
736                            let arrow_type = if is_metadata {
737                                datum_to_arrow_type_with_ree(datum)
738                            } else {
739                                field_id_to_mapped_schema_map
740                                    .get(field_id)
741                                    .ok_or(Error::new(
742                                        ErrorKind::Unexpected,
743                                        "could not find field in schema",
744                                    ))?
745                                    .0
746                                    .data_type()
747                                    .clone()
748                            };
749
750                            return Ok(ColumnSource::Add {
751                                value: Some(datum.literal().clone()),
752                                target_type: arrow_type,
753                            });
754                        }
755                        // Identity partition field present in the file -- fall through
756                        // to read from the file instead of using the constant.
757                    }
758                    Some(ColumnConstant::Null(arrow_type)) => {
759                        return Ok(ColumnSource::Add {
760                            value: None,
761                            target_type: arrow_type.clone(),
762                        });
763                    }
764                    Some(ColumnConstant::CoalesceLastUpdatedSeq(datum)) => {
765                        // Registered only for `_last_updated_sequence_number`, and only after
766                        // the pipeline detects the embedded field id, so the source column is
767                        // always present in the file.
768                        let (_, source_index) =
769                            field_id_to_source_schema_map.get(field_id).ok_or_else(|| {
770                                Error::new(
771                                    ErrorKind::Unexpected,
772                                    format!(
773                                        "coalesce registered for field id {field_id} but the \
774                                         column is absent from the file batch"
775                                    ),
776                                )
777                            })?;
778                        return Ok(ColumnSource::CoalesceLastUpdatedSeq {
779                            source_index: *source_index,
780                            fallback: datum.literal().clone(),
781                            target_type: datum_to_arrow_type_with_ree(datum),
782                        });
783                    }
784                    None => {}
785                }
786
787                if virtual_fields.contains(field_id) {
788                    let (_, source_index) = field_id_to_source_schema_map
789                        .get(field_id)
790                        .ok_or_else(|| Error::new(
791                            ErrorKind::Unexpected,
792                            format!("virtual column for field id {field_id} is not present in source batch — ensure ArrowReaderOptions::with_virtual_columns was applied"),
793                        ))?;
794                    return Ok(ColumnSource::PassThrough { source_index: *source_index });
795                }
796
797                let (target_field, _) =
798                    field_id_to_mapped_schema_map
799                        .get(field_id)
800                        .ok_or(Error::new(
801                            ErrorKind::Unexpected,
802                            format!("could not find field {field_id} in schema"),
803                        ))?;
804                let target_type = target_field.data_type();
805
806                let iceberg_field = snapshot_schema.field_by_id(*field_id).ok_or(Error::new(
807                    ErrorKind::Unexpected,
808                    "Field not found in snapshot schema",
809                ))?;
810
811                // Iceberg spec's "Column Projection" rules (https://iceberg.apache.org/spec/#column-projection).
812                // For fields "not present" in data files:
813                // 1. Use partition metadata (identity transforms only)
814                // 2. Use name mapping
815                // 3. Use initial_default
816                // 4. Return null
817                //
818                // Why check partition constants before Parquet field IDs (Java: BaseParquetReaders.java:299):
819                // In add_files scenarios, partition columns may exist in BOTH Parquet AND partition metadata.
820                // Partition metadata is authoritative - it defines which partition this file belongs to.
821
822                // Field ID resolution now happens in ArrowReader via:
823                // 1. Embedded field IDs (ParquetSchemaUtil.hasIds() = true) - trust them
824                // 2. Name mapping (ParquetSchemaUtil.applyNameMapping()) - applied upfront
825                // 3. Position-based fallback (ParquetSchemaUtil.addFallbackIds()) - applied upfront
826                //
827                // At this point, all field IDs in the source schema are trustworthy.
828                // No conflict detection needed - schema resolution happened in reader.rs.
829                let field_by_id = field_id_to_source_schema_map.get(field_id).map(
830                    |(source_field, source_index)| {
831                        if source_field.data_type().equals_datatype(target_type) {
832                            ColumnSource::PassThrough {
833                                source_index: *source_index,
834                            }
835                        } else {
836                            ColumnSource::Promote {
837                                target_type: target_type.clone(),
838                                source_index: *source_index,
839                            }
840                        }
841                    },
842                );
843
844                // Apply spec's fallback steps for "not present" fields.
845                // Rule #1 (constants) is handled at the beginning of this function
846                let column_source = if let Some(source) = field_by_id {
847                    source
848                } else {
849                    // Rules #2, #3 and #4:
850                    // Rule #2 (name mapping) was already applied in reader.rs if needed.
851                    // If field_id is still not found, the column doesn't exist in the Parquet file.
852                    // Fall through to rule #3 (initial_default) or rule #4 (null).
853                    //
854                    // Per the spec's "Default values", null is only a valid default for an
855                    // optional field. A required field that is absent with no initial-default
856                    // therefore has no valid value and must error, rather than producing a null
857                    // column that violates the field's required constraint. This matches
858                    // Iceberg-Java's Parquet readers (BaseParquetReaders / SparkParquetReaders),
859                    // which raise "Missing required field: <name>".
860                    if iceberg_field.initial_default.is_none() && iceberg_field.required {
861                        return Err(Error::new(
862                            ErrorKind::DataInvalid,
863                            format!("Missing required field: {}", iceberg_field.name),
864                        ));
865                    }
866
867                    let default_value = iceberg_field.initial_default.as_ref().and_then(|lit| {
868                        if let Literal::Primitive(prim) = lit {
869                            Some(prim.clone())
870                        } else {
871                            None
872                        }
873                    });
874
875                    ColumnSource::Add {
876                        value: default_value,
877                        target_type: target_type.clone(),
878                    }
879                };
880
881                Ok(column_source)
882            })
883            .collect()
884    }

886    fn build_field_id_to_arrow_schema_map(
887        source_schema: &SchemaRef,
888    ) -> Result<HashMap<i32, (FieldRef, usize)>> {
889        let mut field_id_to_source_schema = HashMap::new();
890        for (source_field_idx, source_field) in source_schema.fields.iter().enumerate() {
891            // Check if field has a field ID in metadata
892            if let Some(field_id_str) = source_field.metadata().get(PARQUET_FIELD_ID_META_KEY) {
893                let this_field_id = field_id_str.parse().map_err(|e| {
894                    Error::new(
895                        ErrorKind::DataInvalid,
896                        format!("field id not parseable as an i32: {e}"),
897                    )
898                })?;
899
900                field_id_to_source_schema
901                    .insert(this_field_id, (source_field.clone(), source_field_idx));
902            }
903            // If field doesn't have a field ID, skip it - name mapping will handle it
904        }
905
906        Ok(field_id_to_source_schema)
907    }

910        &self,
911        columns: &[Arc<dyn ArrowArray>],
912        operations: &[ColumnSource],
913    ) -> Result<Vec<Arc<dyn ArrowArray>>> {
914        if columns.is_empty() {
915            return Ok(columns.to_vec());
916        }
917        let num_rows = columns[0].len();
918
919        operations
920            .iter()
921            .map(|op| {
922                Ok(match op {
923                    ColumnSource::PassThrough { source_index } => columns[*source_index].clone(),
924
925                    ColumnSource::Promote {
926                        target_type,
927                        source_index,
928                    } => cast(&*columns[*source_index], target_type)?,
929
930                    ColumnSource::Add { target_type, value } => {
931                        Self::create_column(target_type, value, num_rows)?
932                    }
933
934                    ColumnSource::AddStructConstant {
935                        fields,
936                        child_values,
937                    } => Self::create_struct_column(fields, child_values, num_rows)?,
938
939                    ColumnSource::CoalesceLastUpdatedSeq {
940                        source_index,
941                        fallback,
942                        target_type,
943                    } => Self::create_coalesce_column(
944                        &columns[*source_index],
945                        fallback,
946                        target_type,
947                    )?,
948                })
949            })
950            .collect()
951    }

953    /// Builds a coalesced column: the source column's per-row value where non-null,
954    /// else the scalar `fallback`, cast to `target_type` (the run-end-encoded type
955    /// the column's other paths produce).
956    fn create_coalesce_column(
957        source: &ArrayRef,
958        fallback: &PrimitiveLiteral,
959        target_type: &DataType,
960    ) -> Result<ArrayRef> {
961        if source.data_type() != &DataType::Int64 {
962            return Err(Error::new(
963                ErrorKind::DataInvalid,
964                format!("coalesce source must be Int64, got {}", source.data_type()),
965            ));
966        }
967        let PrimitiveLiteral::Long(seq) = fallback else {
968            return Err(Error::new(
969                ErrorKind::Unexpected,
970                format!("coalesce fallback must be a long, got {fallback:?}"),
971            ));
972        };
973        let scalar = Int64Array::new_scalar(*seq);
974        let mask = is_not_null(source)?;
975        let coalesced = zip(&mask, source, &scalar)?;
976        // Cast to the run-end-encoded target type for column-type uniformity across the
977        // constant/null/coalesce paths (not for compression -- a per-row-varying column
978        // gains nothing from REE).
979        Ok(cast(&coalesced, target_type)?)
980    }

982    fn create_column(
983        target_type: &DataType,
984        prim_lit: &Option<PrimitiveLiteral>,
985        num_rows: usize,
986    ) -> Result<ArrayRef> {
987        // Check if this is a RunEndEncoded type (for constant fields)
988        if let DataType::RunEndEncoded(_, values_field) = target_type {
989            // Helper to create a Run-End Encoded array
990            let create_ree_array = |values_array: ArrayRef| -> Result<ArrayRef> {
991                let run_ends = if num_rows == 0 {
992                    Int32Array::from(Vec::<i32>::new())
993                } else {
994                    Int32Array::from(vec![num_rows as i32])
995                };
996                Ok(Arc::new(
997                    RunArray::try_new(&run_ends, &values_array).map_err(|e| {
998                        Error::new(
999                            ErrorKind::Unexpected,
1000                            "Failed to create RunArray for constant value",
1001                        )
1002                        .with_source(e)
1003                    })?,
1004                ))
1005            };
1006
1007            // Create the values array using the helper function
1008            let values_array =
1009                create_primitive_array_single_element(values_field.data_type(), prim_lit)?;
1010
1011            // Wrap in Run-End Encoding
1012            create_ree_array(values_array)
1013        } else {
1014            // Non-REE type (simple arrays for non-constant fields)
1015            create_primitive_array_repeated(target_type, prim_lit, num_rows)
1016        }
1017    }

1048/// Builds a [`StructConstant`] from the unified partition type and a file's
1049/// partition spec/data.
1050///
1051/// If `unified_partition_type` has no fields (unpartitioned table), returns an empty constant
1052/// that renders as a null struct column.
1053///
1054/// For each field in the unified partition type:
1055/// - If it corresponds to a field in this file's partition spec, use the value from partition_data
1056/// - Otherwise (partition evolution), use null
1057pub(crate) fn build_partition_constant(
1058    unified_partition_type: &StructType,
1059    partition_spec: &PartitionSpec,
1060    partition_data: &Struct,
1061) -> Result<StructConstant> {
1062    // Unpartitioned table: empty struct rendered as null
1063    if unified_partition_type.fields().is_empty() {
1064        return StructConstant::new(Fields::empty(), vec![]);
1065    }
1066
1067    use crate::arrow::type_to_arrow_type;
1068
1069    let spec_fields = partition_spec.fields();
1070
1071    let mut arrow_fields = Vec::with_capacity(unified_partition_type.fields().len());
1072    let mut child_values = Vec::with_capacity(unified_partition_type.fields().len());
1073
1074    for unified_field in unified_partition_type.fields() {
1075        let arrow_type = type_to_arrow_type(&unified_field.field_type)?;
1076        // Don't attach PARQUET_FIELD_ID_META_KEY metadata to child fields --
1077        // Spark's output schema for _partition doesn't include it, and Arrow's
1078        // Field::eq checks metadata, causing the schema adapter to insert an
1079        // unnecessary cast operation per batch.
1080        // Always nullable: partition evolution means any field may be absent
1081        // for files written under a different spec.
1082        let arrow_field = Field::new(&unified_field.name, arrow_type, true);
1083        arrow_fields.push(Arc::new(arrow_field));
1084
1085        // Find matching field in this file's partition spec by field_id
1086        let value = spec_fields
1087            .iter()
1088            .position(|f| f.field_id == unified_field.id)
1089            .and_then(|pos| {
1090                // Get the value from partition_data at this position
1091                match &partition_data[pos] {
1092                    Some(Literal::Primitive(prim)) => Some(prim.clone()),
1093                    _ => None,
1094                }
1095            });
1096
1097        child_values.push(value);
1098    }
1099
1100    StructConstant::new(Fields::from(arrow_fields), child_values)
1101}
